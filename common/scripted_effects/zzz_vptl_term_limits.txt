vptl_record_presidential_term_for_current_ruler = {
	if = {
		limit = {
			NOT = { has_variable = vptl_presidential_transition_applied }
		}
		set_variable = vptl_presidential_transition_applied
		vptl_assign_presidential_order_number_to_current_ruler = yes

	ruler ?= {
		vptl_set_presidential_accession_elected = yes
		if = {
			limit = {
				NOT = { has_variable = vptl_presidential_term_limited }
				has_variable = vptl_presidential_terms_served
			}
			change_variable = { name = vptl_presidential_terms_served add = 1 }
		}
		else_if = {
			limit = {
				NOT = { has_variable = vptl_presidential_term_limited }
			}
			set_variable = { name = vptl_presidential_terms_served value = 1 }
		}

		if = {
			limit = {
				NOT = { has_variable = vptl_presidential_term_limited }
				var:vptl_presidential_terms_served >= 2
			}
			set_variable = vptl_presidential_term_limited
			remove_modifier = modifier_vptl_presidential_first_term
			remove_modifier = modifier_vptl_presidential_second_term
			add_modifier = {
				name = modifier_vptl_presidential_second_term
				years = 4
			}
		}
		else_if = {
			limit = {
				NOT = { has_variable = vptl_presidential_term_limited }
			}
			remove_modifier = modifier_vptl_presidential_first_term
			remove_modifier = modifier_vptl_presidential_second_term
			add_modifier = {
				name = modifier_vptl_presidential_first_term
				years = 4
			}
		}
		vptl_refresh_presidential_office_markers = yes
		remove_variable = vptl_presidential_succeeded_to_office
	}
	}
}

vptl_post_presidential_transition_notification = {
	if = {
		limit = {
			is_player = yes
			ruler ?= {
				is_character_alive = yes
				has_variable = vptl_presidential_order_number
			}
			OR = {
				NOT = { has_variable = vptl_presidential_last_transition_notification_ruler }
				ruler ?= {
					NOT = { owner.var:vptl_presidential_last_transition_notification_ruler ?= this }
				}
			}
		}
		save_scope_as = vptl_presidential_transition_country_scope
		ruler ?= {
			save_scope_as = vptl_presidential_transition_ruler_scope
		}
		if = {
			limit = { has_variable = vptl_presidential_debug_logging }
			debug_log = "BPR checkpoint=post_notification source=vptl_presidential_transition [TimeKeeper.GetCurrentDate.GetString] country=[THIS.GetCountry.GetNameNoFormatting]"
		}
		post_notification = vptl_presidential_transition
		set_variable = { name = vptl_presidential_last_transition_notification_ruler value = scope:vptl_presidential_transition_ruler_scope }
		set_variable = { name = vptl_presidential_transition_notification_cooldown months = 2 }
	}
}

# Party scope. Campaign-generation markers prevent a dissolved and later revived party
# from exposing nominees saved for an older election.
vptl_clear_presidential_party_ticket = {
	remove_variable = vptl_presidential_party_ticket_candidate
	remove_variable = vptl_presidential_party_ticket_running_mate
	remove_variable = vptl_presidential_party_ticket_generation_a
	remove_variable = vptl_presidential_party_ticket_generation_b

# Party scope. Preserve a nomination while both the character and party relationship
# remain valid; only the affected slot is rebuilt.
vptl_maintain_presidential_party_ticket = {
	save_scope_as = vptl_presidential_campaign_party
	# Party scopes have no owner link; root remains the country that invoked the
	# campaign-maintenance effect.
	root = { save_scope_as = vptl_presidential_campaign_country }

	if = {
		limit = {
			has_variable = vptl_presidential_party_ticket_candidate
			var:vptl_presidential_party_ticket_candidate ?= {
				OR = {
					NOT = { vptl_presidential_candidate_eligible = yes }
					NOT = {
						this.interest_group ?= {
							is_member_of_party = scope:vptl_presidential_campaign_party
						}
					}
				}
			}
		}
		remove_variable = vptl_presidential_party_ticket_candidate
	}
	if = {
		limit = {
			has_variable = vptl_presidential_party_ticket_running_mate
			var:vptl_presidential_party_ticket_running_mate ?= {
				OR = {
					NOT = { vptl_presidential_party_existing_running_mate_eligible = yes }
					NOT = {
						this.interest_group ?= {
							is_member_of_party = scope:vptl_presidential_campaign_party
						}
					}
					owner.var:vptl_presidential_final_elected_ruler ?= this
				}
			}
		}
		remove_variable = vptl_presidential_party_ticket_running_mate
	}
	if = {
		limit = {
			has_variable = vptl_presidential_party_ticket_candidate
			has_variable = vptl_presidential_party_ticket_running_mate
			var:vptl_presidential_party_ticket_candidate ?= {
				scope:vptl_presidential_campaign_party.var:vptl_presidential_party_ticket_running_mate ?= this
			}
		}
		remove_variable = vptl_presidential_party_ticket_running_mate
	}

	if = {
		limit = { NOT = { has_variable = vptl_presidential_party_ticket_candidate } }
		scope:vptl_presidential_campaign_country = {
			ruler ?= {
				if = {
					limit = {
						vptl_presidential_candidate_eligible = yes
						NOT = { owner.var:vptl_presidential_final_elected_ruler ?= this }
						this.interest_group ?= {
							is_member_of_party = scope:vptl_presidential_campaign_party
						}
					}
					save_scope_as = vptl_presidential_party_candidate_scope
				}
			}
		}
		if = {
			limit = {
				NOT = { exists = scope:vptl_presidential_party_candidate_scope }
				any_member = {
					leader ?= {
						vptl_presidential_candidate_eligible = yes
						NOT = { scope:vptl_presidential_campaign_party.var:vptl_presidential_party_ticket_running_mate ?= this }
					}
				}
			}
			ordered_member = {
				limit = {
					leader ?= {
						vptl_presidential_candidate_eligible = yes
						NOT = { scope:vptl_presidential_campaign_party.var:vptl_presidential_party_ticket_running_mate ?= this }
					}
				}
				order_by = ig_clout
				position = 0
				leader ?= { save_scope_as = vptl_presidential_party_candidate_scope }
			}
		}
		if = {
			limit = {
				NOT = { exists = scope:vptl_presidential_party_candidate_scope }
				scope:vptl_presidential_campaign_country = {
					any_scope_character = {
						vptl_presidential_candidate_eligible = yes
						NOT = { scope:vptl_presidential_campaign_party.var:vptl_presidential_party_ticket_running_mate ?= this }
						this.interest_group ?= {
							is_member_of_party = scope:vptl_presidential_campaign_party
						}
					}
				}
			}
			scope:vptl_presidential_campaign_country = {
				ordered_scope_character = {
					limit = {
						vptl_presidential_candidate_eligible = yes
						NOT = { scope:vptl_presidential_campaign_party.var:vptl_presidential_party_ticket_running_mate ?= this }
						this.interest_group ?= {
							is_member_of_party = scope:vptl_presidential_campaign_party
						}
					}
					order_by = prominence
					position = 0
					save_scope_as = vptl_presidential_party_candidate_scope
				}
			}
		}
		if = {
			limit = { exists = scope:vptl_presidential_party_candidate_scope }
			set_variable = { name = vptl_presidential_party_ticket_candidate value = scope:vptl_presidential_party_candidate_scope }
		}
	}

	if = {
		limit = { NOT = { has_variable = vptl_presidential_party_ticket_running_mate } }
		scope:vptl_presidential_campaign_country = {
			if = {
				limit = {
					has_variable = vptl_presidential_successor
					var:vptl_presidential_successor ?= {
						vptl_presidential_new_successor_candidate_eligible = yes
						NOT = { scope:vptl_presidential_campaign_party.var:vptl_presidential_party_ticket_candidate ?= this }
						this.interest_group ?= {
							is_member_of_party = scope:vptl_presidential_campaign_party
						}
					}
				}
				var:vptl_presidential_successor ?= { save_scope_as = vptl_presidential_party_running_mate_scope }
			}
			if = {
				limit = {
					NOT = { exists = scope:vptl_presidential_party_running_mate_scope }
					any_scope_character = {
						vptl_presidential_new_successor_candidate_eligible = yes
						NOT = { scope:vptl_presidential_campaign_party.var:vptl_presidential_party_ticket_candidate ?= this }
						this.interest_group ?= {
							is_member_of_party = scope:vptl_presidential_campaign_party
						}
					}
				}
				ordered_scope_character = {
					limit = {
						vptl_presidential_new_successor_candidate_eligible = yes
						NOT = { scope:vptl_presidential_campaign_party.var:vptl_presidential_party_ticket_candidate ?= this }

		if = {
			limit = { exists = scope:vptl_presidential_party_running_mate_scope }
			set_variable = { name = vptl_presidential_party_ticket_running_mate value = scope:vptl_presidential_party_running_mate_scope }
		}
	}
}

# Country scope. Preserve legacy tickets when an old campaign is first upgraded.
vptl_migrate_legacy_presidential_campaign_tickets = {
	if = {
		limit = { has_variable = vptl_presidential_ticket_candidate }
		var:vptl_presidential_ticket_candidate ?= {
			this.interest_group ?= { party ?= { save_scope_as = vptl_legacy_primary_ticket_party } }
		}
		if = {
			limit = { exists = scope:vptl_legacy_primary_ticket_party }
			scope:vptl_legacy_primary_ticket_party = {
				set_variable = { name = vptl_presidential_party_ticket_candidate value = root.var:vptl_presidential_ticket_candidate }
				if = {
					limit = {
						root = { has_variable = vptl_presidential_ticket_running_mate }
						root.var:vptl_presidential_ticket_running_mate ?= {
							this.interest_group ?= { is_member_of_party = scope:vptl_legacy_primary_ticket_party }
						}
					}
					set_variable = { name = vptl_presidential_party_ticket_running_mate value = root.var:vptl_presidential_ticket_running_mate }
				}
			}
		}
	}
	if = {
		limit = { has_variable = vptl_presidential_opposition_candidate }
		var:vptl_presidential_opposition_candidate ?= {
			this.interest_group ?= { party ?= { save_scope_as = vptl_legacy_opposition_ticket_party } }
		}
		if = {
			limit = { exists = scope:vptl_legacy_opposition_ticket_party }
			scope:vptl_legacy_opposition_ticket_party = {
				set_variable = { name = vptl_presidential_party_ticket_candidate value = root.var:vptl_presidential_opposition_candidate }
				if = {
					limit = {
						root = { has_variable = vptl_presidential_opposition_running_mate }
						root.var:vptl_presidential_opposition_running_mate ?= {
							this.interest_group ?= { is_member_of_party = scope:vptl_legacy_opposition_ticket_party }
						}
					}
					set_variable = { name = vptl_presidential_party_ticket_running_mate value = root.var:vptl_presidential_opposition_running_mate }
				}
			}
		}
	}
	# Legacy slots are migration input only. Clearing them here prevents the old
	# incumbent/opposition model from influencing party-owned maintenance.
	remove_variable = vptl_presidential_ticket_candidate
	remove_variable = vptl_presidential_ticket_running_mate
	remove_variable = vptl_presidential_opposition_candidate
	remove_variable = vptl_presidential_opposition_running_mate
}

vptl_refresh_presidential_campaign_party_registry = {
	save_scope_as = vptl_presidential_campaign_country
	if = {
		limit = { has_variable_list = vptl_presidential_campaign_parties }
		every_in_list = {
			variable = vptl_presidential_campaign_parties
			save_scope_as = vptl_previous_campaign_party
			if = {
				limit = {
					NOT = {
						scope:vptl_presidential_campaign_country = {
							any_active_party = { this = scope:vptl_previous_campaign_party }
						}
					}
				}
				scope:vptl_previous_campaign_party = {
					vptl_clear_presidential_party_ticket = yes
				}
			}
		}
		clear_variable_list = vptl_presidential_campaign_parties
	}
	every_active_party = {
		limit = { party_has_members = yes }
		if = {
			limit = { NOT = { vptl_presidential_party_ticket_generation_current = yes } }
			vptl_clear_presidential_party_ticket = yes
			if = {
				limit = {
					scope:vptl_presidential_campaign_country = {
						var:vptl_presidential_campaign_ticket_generation = 1
					}
				}
				set_variable = vptl_presidential_party_ticket_generation_a
			}
			else = { set_variable = vptl_presidential_party_ticket_generation_b }
		}
		save_scope_as = vptl_current_campaign_party
		scope:vptl_presidential_campaign_country = {
			add_to_variable_list = {
				name = vptl_presidential_campaign_parties
				target = scope:vptl_current_campaign_party
			}
		}
	}
}

vptl_clear_presidential_campaign_party_tickets = {
	if = {
		limit = { has_variable_list = vptl_presidential_campaign_parties }
		every_in_list = {
			variable = vptl_presidential_campaign_parties
			vptl_clear_presidential_party_ticket = yes
		}
		clear_variable_list = vptl_presidential_campaign_parties
	}
	remove_variable = vptl_presidential_party_ticket_system_active
}

vptl_update_presidential_ticket_candidates = {
	if = {
		limit = {
			has_variable = vptl_presidential_election_transition
			in_election_campaign = yes
			NOT = { has_variable = vptl_presidential_election_settlement_in_progress }
		}
		if = {
			limit = { NOT = { has_variable = vptl_presidential_party_ticket_system_active } }
			vptl_clear_presidential_campaign_party_tickets = yes
		if = {
			limit = {
				has_variable = vptl_presidential_campaign_ticket_generation
				var:vptl_presidential_campaign_ticket_generation = 1
			}
			set_variable = { name = vptl_presidential_campaign_ticket_generation value = 2 }
		}
			else = { set_variable = { name = vptl_presidential_campaign_ticket_generation value = 1 } }
			set_variable = vptl_presidential_party_ticket_system_active
			vptl_refresh_presidential_campaign_party_registry = yes
			vptl_migrate_legacy_presidential_campaign_tickets = yes
		}
		else = { vptl_refresh_presidential_campaign_party_registry = yes }
		if = {
			limit = { has_variable_list = vptl_presidential_campaign_parties }
			every_in_list = {
				variable = vptl_presidential_campaign_parties
				vptl_maintain_presidential_party_ticket = yes
			}
		}
	}
}

vptl_prepare_presidential_term_limits_for_campaign = {
	# Start every campaign from a fresh party generation, including after an
	# interrupted campaign that never reached normal settlement cleanup.
	vptl_clear_presidential_campaign_party_tickets = yes
	set_variable = { name = vptl_presidential_election_transition months = 10 }
	vptl_begin_presidential_election_ruler_notification_suppression = yes
	remove_variable = vptl_presidential_election_settlement_in_progress
	remove_variable = vptl_presidential_election_settled
	remove_variable = vptl_presidential_vanilla_selected_ruler
	remove_variable = vptl_presidential_final_elected_ruler
	remove_variable = vptl_presidential_final_running_mate
	remove_variable = vptl_presidential_winning_ticket_side
	remove_variable = vptl_presidential_winning_ticket_party
	remove_variable = vptl_presidential_winning_side_ambiguous
	remove_variable = vptl_presidential_transition_applied
	remove_variable = vptl_presidential_successor_term_recorded_for_transition
	remove_variable = vptl_presidential_force_successor_transition
	remove_variable = vptl_presidential_outgoing_ruler
	remove_variable = vptl_presidential_transition_outgoing_ruler
	if = {
		limit = { ruler ?= { is_character_alive = yes } }
		vptl_sync_presidential_current_ruler = yes
		set_variable = { name = vptl_presidential_transition_outgoing_ruler value = ruler }
	}
	else_if = {
		limit = {
			has_variable = vptl_presidential_current_ruler
			var:vptl_presidential_current_ruler ?= { is_character_alive = yes }
		}
		set_variable = { name = vptl_presidential_transition_outgoing_ruler value = var:vptl_presidential_current_ruler }
	}
	remove_variable = vptl_presidential_ticket_candidate
	remove_variable = vptl_presidential_ticket_running_mate
	remove_variable = vptl_presidential_opposition_candidate
	remove_variable = vptl_presidential_opposition_running_mate
	vptl_prefer_historical_presidential_ig_leaders = yes
	vptl_update_presidential_successor = yes

	ruler ?= {
		if = {
			limit = {
				NOT = { has_variable = vptl_presidential_term_limited }
				has_variable = vptl_presidential_terms_served
			}
			if = {
				limit = {
					var:vptl_presidential_terms_served >= 2
				}
				set_variable = vptl_presidential_term_limited
			}
		}
	}

	every_interest_group = {
		limit = {
			leader ?= {
				is_ruler_of_own_country = no
				OR = {
					has_variable = vptl_presidential_term_limited
					AND = {
						owner = { c:USA ?= this }
						NOT = { vptl_presidential_candidate_eligible = yes }
					}
				}
			}
		}
		save_scope_as = vptl_term_limit_ig
		leader ?= {
			if = {
				limit = { has_variable = vptl_presidential_term_limited }
				remove_modifier = modifier_vptl_presidential_term_limited
				add_modifier = {
					name = modifier_vptl_presidential_term_limited
				}
			}
		}
		vptl_replace_invalid_presidential_ig_leader_for_saved_ig = yes
	}
	vptl_update_presidential_successor = yes
	vptl_update_presidential_ticket_candidates = yes
}

			has_variable = vptl_presidential_ticket_candidate
			var:vptl_presidential_ticket_candidate ?= {

					NOT = { is_character_alive = yes }
					NOT = { vptl_presidential_candidate_eligible = yes }
					has_variable = vptl_presidential_term_limited
				}
			}
		}
		remove_variable = vptl_presidential_ticket_candidate

	if = {
		limit = {

					NOT = { is_character_alive = yes }

					has_variable = vptl_presidential_term_limited
					is_ruler_of_own_country = yes
					is_heir_of_own_country = yes
					owner.var:vptl_presidential_ticket_candidate ?= this
				}
			}
		}
		remove_variable = vptl_presidential_opposition_candidate

		limit = {
			has_variable = vptl_presidential_ticket_running_mate

					owner.var:vptl_presidential_opposition_candidate ?= this
					owner.var:vptl_presidential_opposition_running_mate ?= this
				}
			}
		}
		remove_variable = vptl_presidential_ticket_running_mate
	}
	if = {
		limit = {
			has_variable = vptl_presidential_opposition_running_mate
			var:vptl_presidential_opposition_running_mate ?= {
				OR = {
					NOT = { vptl_presidential_existing_campaign_running_mate_eligible = yes }

					owner.var:vptl_presidential_ticket_running_mate ?= this

		}

	}

vptl_identify_presidential_winning_ticket_side_legacy = {
	remove_variable = vptl_presidential_winning_ticket_side
	remove_variable = vptl_presidential_incumbent_party_match
	remove_variable = vptl_presidential_opposition_party_match
	remove_variable = vptl_presidential_incumbent_ig_match
	remove_variable = vptl_presidential_opposition_ig_match
	remove_variable = vptl_presidential_winning_side_ambiguous

	ruler ?= { save_scope_as = vptl_presidential_vanilla_selected_ruler_scope }
	set_variable = { name = vptl_presidential_vanilla_selected_ruler value = scope:vptl_presidential_vanilla_selected_ruler_scope }

	if = {
		limit = {
			has_variable = vptl_presidential_ticket_candidate
			ruler ?= { owner.var:vptl_presidential_ticket_candidate ?= this }
		}
		set_variable = { name = vptl_presidential_winning_ticket_side value = 1 }
	}
	else_if = {
		limit = {
			has_variable = vptl_presidential_opposition_candidate
			ruler ?= { owner.var:vptl_presidential_opposition_candidate ?= this }
		}
		set_variable = { name = vptl_presidential_winning_ticket_side value = 2 }
	}
	else_if = {
		limit = {
			has_variable = vptl_presidential_election_transition
			has_variable = vptl_presidential_death_succession_lock
			has_variable = vptl_presidential_death_successor
			has_variable = vptl_presidential_ticket_running_mate
			ruler ?= {
				owner.var:vptl_presidential_death_successor ?= this
				owner.var:vptl_presidential_ticket_running_mate ?= this
			}
		}
		set_variable = { name = vptl_presidential_winning_ticket_side value = 1 }
	}
	else_if = {
		limit = {
			has_variable = vptl_presidential_election_transition
			has_variable = vptl_presidential_death_succession_lock
			has_variable = vptl_presidential_death_successor
			has_variable = vptl_presidential_opposition_running_mate
			ruler ?= {
				owner.var:vptl_presidential_death_successor ?= this
				owner.var:vptl_presidential_opposition_running_mate ?= this
			}
		}
		set_variable = { name = vptl_presidential_winning_ticket_side value = 2 }
	}
	else = {
		ruler ?= {
			this.interest_group ?= {
				save_scope_as = vptl_presidential_vanilla_selected_ig
				party ?= { save_scope_as = vptl_presidential_vanilla_selected_party }
			}
		}

		if = {
			limit = {
				exists = scope:vptl_presidential_vanilla_selected_party
				has_variable = vptl_presidential_ticket_candidate
				var:vptl_presidential_ticket_candidate ?= {
					this.interest_group ?= {
						is_member_of_party = scope:vptl_presidential_vanilla_selected_party
					}
				}
			}
			set_variable = vptl_presidential_incumbent_party_match
		}
		if = {
			limit = {
				exists = scope:vptl_presidential_vanilla_selected_party
				has_variable = vptl_presidential_opposition_candidate
				var:vptl_presidential_opposition_candidate ?= {
					this.interest_group ?= {
						is_member_of_party = scope:vptl_presidential_vanilla_selected_party
					}
				}
			}
			set_variable = vptl_presidential_opposition_party_match
		}
		if = {
			limit = {
				has_variable = vptl_presidential_incumbent_party_match
				NOT = { has_variable = vptl_presidential_opposition_party_match }
			}
			set_variable = { name = vptl_presidential_winning_ticket_side value = 1 }
		}
		else_if = {
			limit = {
				has_variable = vptl_presidential_opposition_party_match
				NOT = { has_variable = vptl_presidential_incumbent_party_match }
			}
			set_variable = { name = vptl_presidential_winning_ticket_side value = 2 }
		}

		if = {
			limit = {
				NOT = { has_variable = vptl_presidential_winning_ticket_side }
				exists = scope:vptl_presidential_vanilla_selected_ig
				has_variable = vptl_presidential_ticket_candidate
				var:vptl_presidential_ticket_candidate ?= {
					this.interest_group ?= {
						is_same_interest_group_type = scope:vptl_presidential_vanilla_selected_ig
					}
				}
			}
			set_variable = vptl_presidential_incumbent_ig_match
		}
		if = {
			limit = {
				NOT = { has_variable = vptl_presidential_winning_ticket_side }
				exists = scope:vptl_presidential_vanilla_selected_ig
				has_variable = vptl_presidential_opposition_candidate
				var:vptl_presidential_opposition_candidate ?= {
					this.interest_group ?= {
						is_same_interest_group_type = scope:vptl_presidential_vanilla_selected_ig
					}
				}
			}
			set_variable = vptl_presidential_opposition_ig_match
		}
		if = {
			limit = {
				NOT = { has_variable = vptl_presidential_winning_ticket_side }
				has_variable = vptl_presidential_incumbent_ig_match

vptl_identify_presidential_winning_ticket_side = {
	remove_variable = vptl_presidential_winning_ticket_side
	remove_variable = vptl_presidential_winning_ticket_party
	remove_variable = vptl_presidential_winning_ticket_party_candidate
	remove_variable = vptl_presidential_winning_ticket_ig_match_count
	remove_variable = vptl_presidential_winning_side_ambiguous

	ruler ?= { save_scope_as = vptl_presidential_vanilla_selected_ruler_scope }
	if = {
		limit = { exists = scope:vptl_presidential_vanilla_selected_ruler_scope }
		set_variable = { name = vptl_presidential_vanilla_selected_ruler value = scope:vptl_presidential_vanilla_selected_ruler_scope }
	}

	# Exact presidential nominee identity is authoritative.
	if = {
		limit = {
			exists = scope:vptl_presidential_vanilla_selected_ruler_scope
			has_variable_list = vptl_presidential_campaign_parties
		}
		every_in_list = {
			variable = vptl_presidential_campaign_parties
			limit = {
				has_variable = vptl_presidential_party_ticket_candidate
				var:vptl_presidential_party_ticket_candidate ?= {
					this = scope:vptl_presidential_vanilla_selected_ruler_scope
				}
				owner = { NOT = { has_variable = vptl_presidential_winning_ticket_party } }
			}
			owner = { set_variable = { name = vptl_presidential_winning_ticket_party value = prev } }
		}
	}

	# If constitutional succession made an existing running mate the temporary ruler,
	# classify the exact saved ticket without promoting that running mate on the ticket.
	if = {
		limit = {
			NOT = { has_variable = vptl_presidential_winning_ticket_party }
			has_variable = vptl_presidential_election_transition
			has_variable = vptl_presidential_death_succession_lock
			has_variable = vptl_presidential_death_successor
			has_variable_list = vptl_presidential_campaign_parties
			ruler ?= { owner.var:vptl_presidential_death_successor ?= this }
		}
		every_in_list = {
			variable = vptl_presidential_campaign_parties
			limit = {
				has_variable = vptl_presidential_party_ticket_running_mate
				var:vptl_presidential_party_ticket_running_mate ?= {
					owner.var:vptl_presidential_death_successor ?= this
				}
				owner = { NOT = { has_variable = vptl_presidential_winning_ticket_party } }
			}
			owner = { set_variable = { name = vptl_presidential_winning_ticket_party value = prev } }
		}
	}

	# Party identity is the next strongest signal from vanilla's selected ruler.
	if = {
		limit = {
			NOT = { has_variable = vptl_presidential_winning_ticket_party }
			exists = scope:vptl_presidential_vanilla_selected_ruler_scope
			has_variable_list = vptl_presidential_campaign_parties
		}
		scope:vptl_presidential_vanilla_selected_ruler_scope = {
			this.interest_group ?= { party ?= { save_scope_as = vptl_presidential_vanilla_selected_party } }
		}
		if = {
			limit = {
				exists = scope:vptl_presidential_vanilla_selected_party
				any_in_list = {
					variable = vptl_presidential_campaign_parties
					this = scope:vptl_presidential_vanilla_selected_party
				}
			}
			set_variable = { name = vptl_presidential_winning_ticket_party value = scope:vptl_presidential_vanilla_selected_party }
		}
	}

	# Independent or reshuffled rulers may still identify exactly one party through IG type.
	if = {
		limit = {
			NOT = { has_variable = vptl_presidential_winning_ticket_party }
			exists = scope:vptl_presidential_vanilla_selected_ruler_scope
			has_variable_list = vptl_presidential_campaign_parties
		}
		scope:vptl_presidential_vanilla_selected_ruler_scope = {
			this.interest_group ?= { save_scope_as = vptl_presidential_vanilla_selected_ig }
		}
		if = {
			limit = { exists = scope:vptl_presidential_vanilla_selected_ig }
			set_variable = { name = vptl_presidential_winning_ticket_ig_match_count value = 0 }
			every_in_list = {
				variable = vptl_presidential_campaign_parties
				limit = {
					has_variable = vptl_presidential_party_ticket_candidate
					var:vptl_presidential_party_ticket_candidate ?= {
						this.interest_group ?= {
							is_same_interest_group_type = scope:vptl_presidential_vanilla_selected_ig
						}
					}
				}
				owner = {
					change_variable = { name = vptl_presidential_winning_ticket_ig_match_count add = 1 }
					set_variable = { name = vptl_presidential_winning_ticket_party_candidate value = prev }
				}
			}
			if = {
				limit = { var:vptl_presidential_winning_ticket_ig_match_count = 1 }
				set_variable = { name = vptl_presidential_winning_ticket_party value = var:vptl_presidential_winning_ticket_party_candidate }
			}
		}
	}

	if = {
		limit = { NOT = { has_variable = vptl_presidential_winning_ticket_party } }
		set_variable = { name = vptl_presidential_winning_side_ambiguous days = 30 }
	}
	remove_variable = vptl_presidential_winning_ticket_party_candidate
	remove_variable = vptl_presidential_winning_ticket_ig_match_count
}

vptl_repair_presidential_elect_ticket = {
	if = {
		limit = {
			OR = {
				NOT = { has_variable = vptl_president_elect }
				var:vptl_president_elect ?= { NOT = { is_character_alive = yes } }
				var:vptl_president_elect ?= { NOT = { vptl_presidential_candidate_eligible = yes } }
			}
		}
		if = {
			limit = {
				has_variable = vptl_vice_president_elect
				var:vptl_vice_president_elect ?= { vptl_presidential_candidate_eligible = yes }
			}
			set_variable = { name = vptl_president_elect value = var:vptl_vice_president_elect }
			remove_variable = vptl_vice_president_elect
		}
		else = { vptl_select_transition_replacement_president_elect = yes }
		if = {
			limit = { has_variable = vptl_president_elect }
			vptl_select_transition_replacement_vice_president_elect = yes
		}
	}
	else_if = {
		limit = {
			OR = {
				NOT = { has_variable = vptl_vice_president_elect }
				var:vptl_vice_president_elect ?= { NOT = { is_character_alive = yes } }
				var:vptl_vice_president_elect ?= {
					NOT = {
						OR = {
							vptl_presidential_transition_replacement_running_mate_eligible = yes
							vptl_presidential_certified_outgoing_vice_president_elect_eligible = yes
						}
					}
				}
			}
		}
		vptl_select_transition_replacement_vice_president_elect = yes
	}
	if = {
		limit = { NOT = { has_variable = vptl_president_elect } }
		vptl_post_unusual_presidential_transition_notification = yes
		vptl_cleanup_presidential_inauguration_transition = yes
	}
	else = { vptl_refresh_presidential_elect_roles = yes }
}

			var:vptl_presidential_transition_outgoing_ruler ?= { is_character_alive = yes }
			ruler ?= { NOT = { owner.var:vptl_presidential_transition_outgoing_ruler ?= this } }
		}
		vptl_suppress_next_presidential_ruler_notification = yes
		var:vptl_presidential_transition_outgoing_ruler ?= {
			if = {
				limit = { owner ?= { has_variable = vptl_presidential_debug_logging } }
				debug_log = "BPR checkpoint=set_character_as_ruler source=inauguration_outgoing_restore target=[THIS.GetCharacter.GetFullNameNoFormatting]"
			}
			set_character_as_ruler = yes
		}
		vptl_sync_presidential_current_ruler = yes
	}
}

vptl_restore_transition_acting_ruler = {
	if = {

			has_variable = vptl_presidential_current_ruler
			var:vptl_presidential_current_ruler ?= { is_character_alive = yes }
			NOT = { has_variable = vptl_presidential_inauguration_in_progress }
		}
		if = {
			limit = { ruler ?= { NOT = { owner.var:vptl_presidential_current_ruler ?= this } } }
			vptl_suppress_next_presidential_ruler_notification = yes
			var:vptl_presidential_current_ruler ?= {
				if = {
					limit = { owner ?= { has_variable = vptl_presidential_debug_logging } }
					debug_log = "BPR checkpoint=set_character_as_ruler source=inauguration_acting_restore target=[THIS.GetCharacter.GetFullNameNoFormatting]"
				}
				set_character_as_ruler = yes
			}
			vptl_sync_presidential_current_ruler = yes
		}
		set_variable = { name = vptl_presidential_transition_outgoing_ruler value = var:vptl_presidential_current_ruler }
	}
}

vptl_schedule_presidential_inauguration = {
	if = {
		limit = { NOT = { has_variable = vptl_presidential_inauguration_schedule_initialized } }
		set_variable = vptl_presidential_inauguration_schedule_initialized
		remove_variable = vptl_presidential_inauguration_timer
		remove_variable = vptl_presidential_inauguration_due
		if = {
			limit = { var:vptl_presidential_inauguration_rule > 0 }
			if = {
				limit = {
					has_variable = vptl_presidential_inauguration_journal_generation
					var:vptl_presidential_inauguration_journal_generation = 1
				}
				set_variable = { name = vptl_presidential_inauguration_journal_generation value = 2 }
			}
			else = {
				set_variable = { name = vptl_presidential_inauguration_journal_generation value = 1 }
			}
			set_variable = vptl_presidential_inauguration_journal_generation_assigned
			if = {
				limit = { var:vptl_presidential_inauguration_rule = 30 }
				set_variable = { name = vptl_presidential_inauguration_timer days = 30 }
			}
			else_if = {
				limit = { var:vptl_presidential_inauguration_rule = 60 }
				set_variable = { name = vptl_presidential_inauguration_timer days = 60 }
			}
			else_if = {
				limit = { var:vptl_presidential_inauguration_rule = 90 }
				set_variable = { name = vptl_presidential_inauguration_timer days = 90 }
			}
			else_if = {
				limit = { var:vptl_presidential_inauguration_rule = 120 }
				set_variable = { name = vptl_presidential_inauguration_timer days = 120 }
			}
			vptl_ensure_presidential_transition_journal = yes
		}
		else = {
			# Rule 0 is pending and immediately due; it uses the same inauguration path synchronously.
			set_variable = vptl_presidential_inauguration_due
			vptl_inaugurate_president_elect = yes
		}
	}
}

vptl_inaugurate_president_elect = {
	if = {
		limit = {
			vptl_presidential_transition_qualifies = yes
			vptl_presidential_inauguration_is_due = yes
			NOT = { has_variable = vptl_presidential_inauguration_in_progress }
			NOT = { has_variable = vptl_presidential_inauguration_completed }
		}
		vptl_repair_presidential_elect_ticket = yes
		if = {
			limit = {
				has_variable = vptl_presidential_inauguration_pending
				has_variable = vptl_president_elect
				var:vptl_president_elect ?= { vptl_presidential_candidate_eligible = yes }
			}
			set_variable = vptl_presidential_inauguration_in_progress
			if = {
				limit = {
					has_variable = vptl_presidential_transition_outgoing_ruler
					var:vptl_presidential_transition_outgoing_ruler ?= {
						is_character_alive = yes
						NOT = { owner.var:vptl_president_elect ?= this }
						NOT = { has_role = character_role_vptl_former_president }
					}
				}
				var:vptl_presidential_transition_outgoing_ruler ?= {
					add_character_role = character_role_vptl_former_president
				}
			}
			if = {
				limit = { ruler ?= { NOT = { owner.var:vptl_president_elect ?= this } } }
				vptl_suppress_next_presidential_ruler_notification = yes
				var:vptl_president_elect ?= {
					vptl_set_presidential_accession_elected = yes
					vptl_clean_current_president_roles = yes
					set_character_as_ruler = yes
				}
			}
			else = { ruler ?= { vptl_set_presidential_accession_elected = yes } }
			vptl_sync_presidential_current_ruler = yes
			vptl_assign_presidential_order_number_to_current_ruler = yes
			remove_variable = vptl_presidential_successor
			if = {
				limit = {
					has_variable = vptl_vice_president_elect
					var:vptl_vice_president_elect ?= {
						OR = {
							vptl_presidential_new_successor_candidate_eligible = yes
							vptl_presidential_certified_outgoing_vice_president_elect_eligible = yes
						}
						NOT = { owner.var:vptl_president_elect ?= this }
					}
				}
				set_variable = { name = vptl_presidential_successor value = var:vptl_vice_president_elect }
			}
			else = { vptl_update_presidential_successor_unrestricted = yes }
			vptl_refresh_presidential_successor_display_marker = yes
			vptl_record_presidential_term_for_current_ruler = yes
			vptl_record_vice_presidential_term_for_current_successor = yes
			set_variable = vptl_presidential_inauguration_completed
			remove_modifier = modifier_vptl_lame_duck_administration
			remove_variable = vptl_presidential_inauguration_pending
			vptl_refresh_presidential_elect_roles = yes
			vptl_post_presidential_inauguration_notification = yes
			vptl_clear_presidential_transition_variables = yes
		}
	}
}

vptl_certify_presidential_election = {
	if = {
		limit = { has_variable = vptl_presidential_final_elected_ruler }
		vptl_select_final_running_mate_from_winning_ticket = yes
		set_variable = { name = vptl_president_elect value = var:vptl_presidential_final_elected_ruler }
		if = {
			limit = { has_variable = vptl_presidential_final_running_mate }
			set_variable = { name = vptl_vice_president_elect value = var:vptl_presidential_final_running_mate }
		}
		var:vptl_president_elect ?= {
			this.interest_group ?= {
				save_scope_as = vptl_presidential_transition_winning_ig_scope
				party ?= { save_scope_as = vptl_presidential_transition_winning_party_scope }
			}
		}
		if = {
			limit = { exists = scope:vptl_presidential_transition_winning_ig_scope }
			set_variable = { name = vptl_presidential_transition_winning_ig value = scope:vptl_presidential_transition_winning_ig_scope }
		}
		if = {
			limit = { has_variable = vptl_presidential_winning_ticket_party }
			set_variable = { name = vptl_presidential_transition_winning_party value = var:vptl_presidential_winning_ticket_party }
		}
		else_if = {
			limit = { exists = scope:vptl_presidential_transition_winning_party_scope }
			set_variable = { name = vptl_presidential_transition_winning_party value = scope:vptl_presidential_transition_winning_party_scope }
		}
		vptl_mark_losing_presidential_party_tickets = yes
		set_variable = vptl_presidential_inauguration_pending
		set_variable = { name = vptl_presidential_election_settled days = 30 }
		# Certification is authoritative before any ruler restoration can fire a nested callback.
		set_variable = vptl_presidential_certification_complete
		remove_variable = vptl_presidential_transition_incumbent_reelected
		if = {
			limit = {
				OR = {
					NOT = { has_variable = vptl_presidential_transition_outgoing_ruler }
					var:vptl_presidential_transition_outgoing_ruler ?= { NOT = { is_character_alive = yes } }
				}
				has_variable = vptl_presidential_current_ruler
				var:vptl_presidential_current_ruler ?= { is_character_alive = yes }
			}
			set_variable = { name = vptl_presidential_transition_outgoing_ruler value = var:vptl_presidential_current_ruler }
		}
		if = {
			limit = {
				has_variable = vptl_presidential_transition_outgoing_ruler
				var:vptl_president_elect ?= { owner.var:vptl_presidential_transition_outgoing_ruler ?= this }
			}
			set_variable = vptl_presidential_transition_incumbent_reelected
		}
		vptl_refresh_presidential_elect_roles = yes
		if = {
			limit = { vptl_presidential_uses_delayed_inauguration = yes }
			vptl_restore_transition_outgoing_ruler = yes
		}
		vptl_finish_presidential_election_ruler_notification_suppression = yes
		vptl_refresh_lame_duck_administration = yes
		vptl_schedule_presidential_inauguration = yes
	}
	else = { vptl_cleanup_presidential_inauguration_transition = yes }
}

vptl_select_transition_sitting_successor_replacement = {
	remove_variable = vptl_presidential_successor
	if = {
		limit = {
			any_interest_group = {
				leader ?= { vptl_presidential_transition_sitting_successor_replacement_eligible = yes }
			}
		}
		ordered_interest_group = {
			limit = {
				leader ?= { vptl_presidential_transition_sitting_successor_replacement_eligible = yes }
			}
			order_by = ig_clout
			position = 0
			leader ?= { save_scope_as = vptl_transition_sitting_successor_scope }
		}
		set_variable = { name = vptl_presidential_successor value = scope:vptl_transition_sitting_successor_scope }
	}
	if = {
		limit = {
			NOT = { has_variable = vptl_presidential_successor }
			any_scope_character = { vptl_presidential_transition_sitting_successor_replacement_eligible = yes }
		}
		ordered_scope_character = {
			limit = { vptl_presidential_transition_sitting_successor_replacement_eligible = yes }
			order_by = prominence
			position = 0
			save_scope_as = vptl_transition_sitting_successor_scope
		}
		set_variable = { name = vptl_presidential_successor value = scope:vptl_transition_sitting_successor_scope }
	}
	vptl_refresh_presidential_successor_display_marker = yes
}

vptl_repair_transition_sitting_successor = {
	if = {
		limit = {
			has_variable = vptl_presidential_inauguration_pending
			OR = {
				NOT = { has_variable = vptl_presidential_successor }
				AND = {
					has_variable = vptl_presidential_successor
					var:vptl_presidential_successor ?= { NOT = { vptl_presidential_sitting_successor_handoff_eligible = yes } }
				}
			}
		}
		vptl_select_transition_sitting_successor_replacement = yes
	}
	else = { vptl_refresh_presidential_successor_display_marker = yes }
}

vptl_handle_presidential_inauguration_character_death = {
	if = {
		limit = {
			has_variable = vptl_presidential_inauguration_pending
			exists = scope:vptl_presidential_dead_character
		}
		set_variable = vptl_presidential_pending_death_handling
		if = {
			limit = {
				has_variable = vptl_presidential_current_ruler
				var:vptl_presidential_current_ruler ?= { this = scope:vptl_presidential_dead_character }
			}
			vptl_promote_presidential_successor_if_tracked_ruler_is_dead = yes
		}
		if = {
			limit = {
				has_variable = vptl_presidential_successor
				var:vptl_presidential_successor ?= { this = scope:vptl_presidential_dead_character }
			}
			vptl_select_transition_sitting_successor_replacement = yes
		}
		if = {
			limit = { ruler ?= { NOT = { is_character_alive = yes } } }
			vptl_promote_presidential_successor_if_ruler_is_dead = yes
		}
		if = {
			limit = {
				OR = {
					AND = {
						has_variable = vptl_president_elect
						var:vptl_president_elect ?= { this = scope:vptl_presidential_dead_character }
					}
					AND = {
						has_variable = vptl_vice_president_elect
						var:vptl_vice_president_elect ?= { this = scope:vptl_presidential_dead_character }
					}
				}
			}
			vptl_repair_presidential_elect_ticket = yes
		}
		if = {
			limit = {
				has_variable = vptl_presidential_inauguration_pending
				has_variable = vptl_presidential_current_ruler
				var:vptl_presidential_current_ruler ?= { is_character_alive = yes }
			}
			set_variable = { name = vptl_presidential_transition_outgoing_ruler value = var:vptl_presidential_current_ruler }
		}
		else_if = {
			limit = {
				has_variable = vptl_presidential_inauguration_pending
				NOT = { has_variable = vptl_presidential_current_ruler }
				ruler ?= { is_character_alive = yes }
			}
			vptl_sync_presidential_current_ruler = yes
			set_variable = { name = vptl_presidential_transition_outgoing_ruler value = ruler }
		}
		vptl_refresh_presidential_elect_roles = yes
		vptl_refresh_lame_duck_administration = yes
		remove_variable = vptl_presidential_pending_death_handling
	}
}

vptl_maintain_presidential_inauguration_transition = {
	vptl_initialize_presidential_constitution_defaults = yes
	if = {
		limit = { has_variable = vptl_presidential_inauguration_pending }
		if = {
			limit = { NOT = { has_variable = vptl_presidential_inauguration_schedule_initialized } }
			# Existing pending saves keep their original timed variable and do not receive a restarted journal.
			set_variable = vptl_presidential_inauguration_schedule_initialized
			set_variable = vptl_presidential_inauguration_legacy_transition
			if = {
				limit = { NOT = { has_variable = vptl_presidential_inauguration_journal_generation } }
				set_variable = { name = vptl_presidential_inauguration_journal_generation value = 1 }
			}
		}
		if = {
			limit = { NOT = { vptl_presidential_transition_qualifies = yes } }
			vptl_cleanup_presidential_inauguration_transition = yes
		}
		else = {
			if = {
				limit = {
					NOT = { has_variable = vptl_presidential_inauguration_in_progress }
					has_variable = vptl_presidential_transition_outgoing_ruler
					var:vptl_presidential_transition_outgoing_ruler ?= { is_character_alive = yes }
				}
				vptl_restore_transition_outgoing_ruler = yes
			}
			else_if = {
				limit = {
					NOT = { has_variable = vptl_presidential_inauguration_in_progress }
					has_variable = vptl_presidential_current_ruler
					var:vptl_presidential_current_ruler ?= { is_character_alive = yes }
				}
				vptl_restore_transition_acting_ruler = yes
			}
			vptl_repair_presidential_elect_ticket = yes
			vptl_repair_transition_sitting_successor = yes
			vptl_refresh_presidential_elect_roles = yes
			vptl_refresh_lame_duck_administration = yes
			vptl_ensure_presidential_transition_journal = yes
			if = {
				limit = { vptl_presidential_inauguration_is_due = yes }
				set_variable = vptl_presidential_inauguration_due
				vptl_inaugurate_president_elect = yes
			}
		}
	}
}

vptl_settle_presidential_election = {
	if = {
		limit = {
			vptl_presidential_transition_qualifies = yes
			has_variable = vptl_presidential_election_transition
			in_election_campaign = no
			NOT = { has_variable = vptl_presidential_election_settlement_in_progress }
			NOT = { has_variable = vptl_presidential_election_settled }
		}
		set_variable = vptl_presidential_election_settlement_in_progress
		vptl_initialize_presidential_constitution_defaults = yes
		if = {
			limit = { has_variable = vptl_presidential_debug_logging }
			debug_log = "BPR checkpoint=vptl_settle_presidential_election entry [TimeKeeper.GetCurrentDate.GetString] country=[THIS.GetCountry.GetNameNoFormatting]"
		}
		vptl_debug_log_presidential_state = yes
		vptl_prefer_historical_presidential_ig_leaders = yes
		vptl_identify_presidential_winning_ticket_side = yes
		vptl_select_final_president_from_winning_ticket = yes
		vptl_certify_presidential_election = yes
		vptl_clear_presidential_campaign_party_tickets = yes
		remove_variable = vptl_presidential_death_succession_lock
		remove_variable = vptl_presidential_death_successor
		remove_variable = vptl_presidential_force_successor_transition
		remove_variable = vptl_presidential_outgoing_ruler
		remove_variable = vptl_presidential_ticket_candidate
		remove_variable = vptl_presidential_ticket_running_mate
		remove_variable = vptl_presidential_opposition_candidate
		remove_variable = vptl_presidential_opposition_running_mate
		remove_variable = vptl_presidential_election_transition
		if = {
			limit = { has_variable = vptl_presidential_debug_logging }
			debug_log = "BPR checkpoint=election_certification_complete [TimeKeeper.GetCurrentDate.GetString] country=[THIS.GetCountry.GetNameNoFormatting]"
		}
		vptl_debug_log_presidential_state = yes
		remove_variable = vptl_presidential_election_settlement_in_progress
	}
}
