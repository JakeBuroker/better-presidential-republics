vptl_record_presidential_term_for_current_ruler = {
	vptl_assign_presidential_order_number_to_current_ruler = yes

	ruler ?= {
		vptl_set_presidential_accession_elected = yes
		if = {
			limit = {
				NOT = { has_variable = vptl_presidential_term_limited }
				has_variable = vptl_presidential_terms_served
			}
			change_variable = { name = vptl_presidential_terms_served add = 1 }
		}
		else_if = {
			limit = {
				NOT = { has_variable = vptl_presidential_term_limited }
			}
			set_variable = { name = vptl_presidential_terms_served value = 1 }
		}

		if = {
			limit = {
				NOT = { has_variable = vptl_presidential_term_limited }
				var:vptl_presidential_terms_served >= 2
			}
			set_variable = vptl_presidential_term_limited
			remove_modifier = modifier_vptl_presidential_first_term
			remove_modifier = modifier_vptl_presidential_second_term
			add_modifier = {
				name = modifier_vptl_presidential_second_term
				years = 4
			}
		}
		else_if = {
			limit = {
				NOT = { has_variable = vptl_presidential_term_limited }
			}
			remove_modifier = modifier_vptl_presidential_first_term
			remove_modifier = modifier_vptl_presidential_second_term
			add_modifier = {
				name = modifier_vptl_presidential_first_term
				years = 4
			}
		}
		vptl_refresh_presidential_office_markers = yes
		remove_variable = vptl_presidential_succeeded_to_office
	}
	vptl_post_presidential_transition_notification = yes
}

vptl_post_presidential_transition_notification = {
	if = {
		limit = {
			is_player = yes
			ruler ?= {
				is_character_alive = yes
				has_variable = vptl_presidential_order_number
			}
		}
		save_scope_as = vptl_presidential_transition_country_scope
		ruler ?= {
			save_scope_as = vptl_presidential_transition_ruler_scope
		}
		post_notification = vptl_presidential_transition
	}
}

vptl_assign_presidential_order_number_to_current_ruler = {
	vptl_seed_usa_presidential_order_numbers = yes
	vptl_seed_1836_presidential_order_numbers = yes

	if = {
		limit = {
			ruler ?= {
				is_character_alive = yes
				NOT = { has_variable = vptl_presidential_order_number }
			}
		}
		if = {
			limit = {
				NOT = { has_variable = vptl_presidential_order_counter }
			}
			set_variable = { name = vptl_presidential_order_counter value = 0 }
		}
		change_variable = { name = vptl_presidential_order_counter add = 1 }
		ruler ?= {
			set_variable = { name = vptl_presidential_order_number value = owner.var:vptl_presidential_order_counter }
			if = {
				limit = {
					NOT = { has_variable = vptl_presidential_years_served }
				}
				set_variable = { name = vptl_presidential_years_served value = 0 }
			}
			vptl_refresh_presidential_office_markers = yes
		}
	}
}

vptl_refresh_presidential_office_markers = {
	remove_modifier = modifier_vptl_presidential_service_partial_term
	remove_modifier = modifier_vptl_presidential_service_one_term
	remove_modifier = modifier_vptl_presidential_service_two_terms
	remove_modifier = modifier_vptl_vice_presidential_service_one_term
	remove_modifier = modifier_vptl_vice_presidential_service_two_terms

	if = {
		limit = {
			has_variable = vptl_presidential_order_number
			NOT = { has_trait = vptl_presidential_history_marker }
		}
		add_trait = vptl_presidential_history_marker
	}

	if = {
		limit = {
			has_variable = vptl_vice_presidential_terms_served
			NOT = { has_trait = vptl_vice_presidential_history_marker }
		}
		add_trait = vptl_vice_presidential_history_marker
	}

	if = {
		limit = {
			has_variable = vptl_presidential_terms_served
			var:vptl_presidential_terms_served >= 2
		}
		add_modifier = { name = modifier_vptl_presidential_service_two_terms }
	}
	else_if = {
		limit = {
			has_variable = vptl_presidential_terms_served
			var:vptl_presidential_terms_served >= 1
		}
		add_modifier = { name = modifier_vptl_presidential_service_one_term }
	}
	else_if = {
		limit = {
			has_variable = vptl_presidential_order_number
		}
		add_modifier = { name = modifier_vptl_presidential_service_partial_term }
	}

	if = {
		limit = {
			has_variable = vptl_vice_presidential_terms_served
			var:vptl_vice_presidential_terms_served >= 2
		}
		add_modifier = { name = modifier_vptl_vice_presidential_service_two_terms }
	}
	else_if = {
		limit = {
			has_variable = vptl_vice_presidential_terms_served
			var:vptl_vice_presidential_terms_served >= 1
		}
		add_modifier = { name = modifier_vptl_vice_presidential_service_one_term }
	}
}

vptl_initialize_presidential_term_limits = {
	if = {
		limit = {
			has_law_or_variant = law_type:law_presidential_republic
			government_transfer_of_power = presidential_elective
			NOT = { has_law_or_variant = law_type:law_single_party_state }
		}
		vptl_assign_presidential_order_number_to_current_ruler = yes

		every_scope_character = {
			limit = {
				has_template = vptl_usa_john_quincy_adams
			}
			set_variable = { name = vptl_presidential_terms_served value = 2 }
			set_variable = vptl_presidential_term_limited
			remove_modifier = modifier_vptl_presidential_first_term
			remove_modifier = modifier_vptl_presidential_second_term
			remove_modifier = modifier_vptl_presidential_term_limited
			add_modifier = {
				name = modifier_vptl_presidential_term_limited
			}
			vptl_refresh_presidential_office_markers = yes
		}
		ruler ?= {
			if = {
				limit = {
					has_variable = is_andrew_jackson
					NOT = { has_variable = vptl_presidential_terms_served }
					NOT = { has_variable = vptl_presidential_term_limited }
				}
				set_variable = { name = vptl_presidential_terms_served value = 2 }
				remove_modifier = modifier_vptl_presidential_first_term
				remove_modifier = modifier_vptl_presidential_second_term
				add_modifier = {
					name = modifier_vptl_presidential_second_term
					months = 12
				}
				vptl_refresh_presidential_office_markers = yes
			}
			else_if = {
				limit = {
					NOT = { has_variable = vptl_presidential_terms_served }
					NOT = { has_variable = vptl_presidential_term_limited }
					NOT = { has_variable = vptl_presidential_succeeded_to_office }
				}
				set_variable = { name = vptl_presidential_terms_served value = 1 }
				remove_modifier = modifier_vptl_presidential_first_term
				remove_modifier = modifier_vptl_presidential_second_term
				add_modifier = {
					name = modifier_vptl_presidential_first_term
					years = 4
				}
				vptl_refresh_presidential_office_markers = yes
			}

			if = {
				limit = {
					has_variable = vptl_presidential_term_limited
				}
				remove_modifier = modifier_vptl_presidential_term_limited
				add_modifier = {
					name = modifier_vptl_presidential_term_limited
				}
			}
			else_if = {
				limit = {
					NOT = { has_variable = vptl_presidential_term_limited }
					has_variable = vptl_presidential_terms_served
					var:vptl_presidential_terms_served >= 2
					NOT = { has_modifier = modifier_vptl_presidential_second_term }
				}
				add_modifier = {
					name = modifier_vptl_presidential_second_term
					years = 4
				}
				vptl_refresh_presidential_office_markers = yes
			}
			else_if = {
				limit = {
					NOT = { has_variable = vptl_presidential_term_limited }
					has_variable = vptl_presidential_terms_served
					var:vptl_presidential_terms_served >= 1
					NOT = { var:vptl_presidential_terms_served >= 2 }
					NOT = { has_modifier = modifier_vptl_presidential_first_term }
				}
				add_modifier = {
					name = modifier_vptl_presidential_first_term
					years = 4
				}
				vptl_refresh_presidential_office_markers = yes
			}
		}
		vptl_prefer_historical_presidential_ig_leaders = yes
	}
	vptl_update_presidential_successor = yes
}

	vptl_update_presidential_successor = yes

	ruler ?= {
		if = {
			limit = {
				NOT = { has_variable = vptl_presidential_term_limited }
				has_variable = vptl_presidential_terms_served
			}
			if = {
				limit = {
					var:vptl_presidential_terms_served >= 2
				}
				set_variable = vptl_presidential_term_limited
			}
		}
		if = {
			limit = {
				has_variable = vptl_presidential_term_limited
			}
			save_scope_as = vptl_presidential_outgoing_ruler_scope
		}
	}
	if = {
		limit = {
			exists = scope:vptl_presidential_outgoing_ruler_scope
			has_variable = vptl_presidential_successor
			var:vptl_presidential_successor ?= {
				is_character_alive = yes
				is_ruler_of_own_country = no
				is_heir_of_own_country = no
				NOT = { has_variable = vptl_presidential_term_limited }
				vptl_presidential_successor_candidate_eligible = yes
			}
		}
		set_variable = { name = vptl_presidential_outgoing_ruler value = scope:vptl_presidential_outgoing_ruler_scope }
		set_variable = { name = vptl_presidential_force_successor_transition months = 10 }
	}

	every_interest_group = {
		limit = {
			leader ?= {
				is_ruler_of_own_country = no
				OR = {
					has_variable = vptl_presidential_term_limited
					AND = {
						owner = { c:USA ?= this }
						NOT = { vptl_presidential_candidate_eligible = yes }
					}
				}
			}
		}
		save_scope_as = vptl_term_limit_ig
		leader ?= {
			if = {
				limit = { has_variable = vptl_presidential_term_limited }
				remove_modifier = modifier_vptl_presidential_term_limited
				add_modifier = {
					name = modifier_vptl_presidential_term_limited
				}
			}
		}
		vptl_set_presidential_successor_as_ig_leader_for_saved_ig = yes
	}
	vptl_update_presidential_successor = yes
}

vptl_apply_presidential_term_limits = {
	if = {
		limit = {
			has_variable = vptl_presidential_force_successor_transition
			has_variable = vptl_presidential_outgoing_ruler
			has_variable = vptl_presidential_successor
			ruler ?= {
				owner.var:vptl_presidential_outgoing_ruler ?= this
			}
			var:vptl_presidential_successor ?= {
				is_character_alive = yes
				is_ruler_of_own_country = no
				is_heir_of_own_country = no
				NOT = { has_variable = vptl_presidential_term_limited }
				vptl_presidential_successor_candidate_eligible = yes
			}
		}
		remove_variable = vptl_presidential_force_successor_transition
		remove_variable = vptl_presidential_outgoing_ruler
		var:vptl_presidential_successor ?= {
			vptl_set_presidential_accession_elected = yes
			set_character_as_ruler = yes
		}
		vptl_initialize_presidential_term_limits = yes
		vptl_post_presidential_transition_notification = yes
	}
	else_if = {
		limit = {
			ruler ?= {
				has_variable = vptl_presidential_term_limited
			}
		}
		vptl_replace_term_limited_presidential_ruler_with_successor = yes
	}
	else_if = {
		limit = {
			c:USA ?= this
			ruler ?= {
				NOT = { vptl_presidential_candidate_eligible = yes }
			}
		}
		vptl_replace_ineligible_usa_presidential_ruler_with_successor = yes
	}
	else_if = {
		limit = {
			ruler ?= {
				NOT = { has_variable = vptl_presidential_term_limited }
				OR = {
					owner = { NOT = { c:USA ?= this } }
					vptl_presidential_candidate_eligible = yes
				}
			}
		}
		vptl_record_presidential_term_for_current_ruler = yes
	}
	vptl_update_presidential_successor = yes
	vptl_record_vice_presidential_term_for_current_successor = yes
	remove_variable = vptl_presidential_election_transition
}
