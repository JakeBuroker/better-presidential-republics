vptl_capture_presidential_history_episode_data = {
	remove_variable = vptl_history_capture_president
	remove_variable = vptl_history_capture_vice_president
	remove_variable = vptl_history_capture_party
	remove_variable = vptl_history_capture_ig
	set_variable = { name = vptl_history_capture_number value = 0 }
	set_variable = { name = vptl_history_capture_accession_type value = 5 }
	set_variable = { name = vptl_history_capture_terms_at_start value = 0 }
	set_variable = { name = vptl_history_capture_gdp_start value = gdp }
	set_variable = { name = vptl_history_capture_prestige_start value = prestige }
	vptl_capture_presidential_history_date = yes
	ruler ?= { save_scope_as = vptl_history_capture_president }
	if = {
		limit = { exists = scope:vptl_history_capture_president }
		set_variable = { name = vptl_history_capture_president value = scope:vptl_history_capture_president }
		if = { limit = { scope:vptl_history_capture_president = { has_variable = vptl_presidential_order_number } } set_variable = { name = vptl_history_capture_number value = scope:vptl_history_capture_president.var:vptl_presidential_order_number } }
		if = { limit = { scope:vptl_history_capture_president = { has_variable = vptl_presidential_terms_served } } set_variable = { name = vptl_history_capture_terms_at_start value = scope:vptl_history_capture_president.var:vptl_presidential_terms_served } }
		if = { limit = { scope:vptl_history_capture_president = { has_variable = vptl_presidential_accession_elected } } set_variable = { name = vptl_history_capture_accession_type value = 1 } }
		else_if = { limit = { scope:vptl_history_capture_president = { has_variable = vptl_presidential_accession_succeeded } } set_variable = { name = vptl_history_capture_accession_type value = 2 } }
		else_if = { limit = { scope:vptl_history_capture_president = { has_variable = vptl_presidential_accession_interim } } set_variable = { name = vptl_history_capture_accession_type value = 3 } }
		else_if = { limit = { scope:vptl_history_capture_president = { has_variable = vptl_presidential_accession_provisional } } set_variable = { name = vptl_history_capture_accession_type value = 4 } }
		scope:vptl_history_capture_president = {
			this.interest_group ?= {
				save_scope_as = vptl_history_capture_ig
				party ?= { save_scope_as = vptl_history_capture_party }
			}
		}
	}
	if = { limit = { exists = scope:vptl_history_capture_ig } set_variable = { name = vptl_history_capture_ig value = scope:vptl_history_capture_ig } }
	if = { limit = { exists = scope:vptl_history_capture_party } set_variable = { name = vptl_history_capture_party value = scope:vptl_history_capture_party } }
	if = {
		limit = { has_variable = vptl_presidential_successor var:vptl_presidential_successor ?= { is_character_alive = yes } }
		var:vptl_presidential_successor ?= { save_scope_as = vptl_history_capture_vice_president }
		set_variable = { name = vptl_history_capture_vice_president value = scope:vptl_history_capture_vice_president }
	}
}

vptl_finalize_presidential_history_archive_slot = {
	if = {
		limit = {
			var:vptl_presidential_history_current_slot = $SLOT$
			has_variable = vptl_presidential_history_slot_$SLOT$_active
			NOT = { has_variable = vptl_presidential_history_slot_$SLOT$_closed }
		}
		if = {
			limit = { has_variable = vptl_presidential_history_current_president var:vptl_presidential_history_current_president ?= { has_variable = vptl_presidential_terms_served } }
			set_variable = { name = vptl_presidential_history_slot_$SLOT$_terms_at_end value = var:vptl_presidential_history_current_president.var:vptl_presidential_terms_served }
		}
		set_variable = { name = vptl_presidential_history_slot_$SLOT$_gdp_end value = gdp }
		set_variable = { name = vptl_presidential_history_slot_$SLOT$_prestige_end value = prestige }
		set_variable = { name = vptl_presidential_history_slot_$SLOT$_departure_reason value = var:vptl_presidential_history_departure_reason }
		set_variable = { name = vptl_presidential_history_slot_$SLOT$_end_year value = var:vptl_history_capture_year }
		set_variable = { name = vptl_presidential_history_slot_$SLOT$_end_month value = var:vptl_history_capture_month }
		set_variable = vptl_presidential_history_slot_$SLOT$_closed
		remove_variable = vptl_presidential_history_slot_$SLOT$_active
	}
}

		else = { remove_variable = vptl_presidential_history_display_$SLOT$_start_year }

		else = { remove_variable = vptl_presidential_history_display_$SLOT$_start_month }

		else = { remove_variable = vptl_presidential_history_display_$SLOT$_end_month }
		if = { limit = { has_variable = vptl_presidential_history_recent_$SLOT$_closed } set_variable = { name = vptl_presidential_history_display_$SLOT$_closed value = 1 } }
		else = { set_variable = { name = vptl_presidential_history_display_$SLOT$_closed value = 0 } }
		set_variable = vptl_presidential_history_display_$SLOT$_populated
	}
	else = {
		remove_variable = vptl_presidential_history_display_$SLOT$_president
		remove_variable = vptl_presidential_history_display_$SLOT$_vice_president
		remove_variable = vptl_presidential_history_display_$SLOT$_party
		remove_variable = vptl_presidential_history_display_$SLOT$_ig
		remove_variable = vptl_presidential_history_display_$SLOT$_number
		remove_variable = vptl_presidential_history_display_$SLOT$_accession_type
		remove_variable = vptl_presidential_history_display_$SLOT$_departure_reason
		remove_variable = vptl_presidential_history_display_$SLOT$_terms_at_start
		remove_variable = vptl_presidential_history_display_$SLOT$_terms_at_end
		remove_variable = vptl_presidential_history_display_$SLOT$_gdp_start
		remove_variable = vptl_presidential_history_display_$SLOT$_gdp_end
		remove_variable = vptl_presidential_history_display_$SLOT$_prestige_start
		remove_variable = vptl_presidential_history_display_$SLOT$_prestige_end
		remove_variable = vptl_presidential_history_display_$SLOT$_start_year
		remove_variable = vptl_presidential_history_display_$SLOT$_start_month
		remove_variable = vptl_presidential_history_display_$SLOT$_end_year
		remove_variable = vptl_presidential_history_display_$SLOT$_end_month
		remove_variable = vptl_presidential_history_display_$SLOT$_closed
		remove_variable = vptl_presidential_history_display_$SLOT$_populated
	}
}

vptl_sync_presidential_history_after_ruler_change = {
	if = {
		limit = { NOT = { has_variable = vptl_presidential_history_sync_in_progress } }
		set_variable = vptl_presidential_history_sync_in_progress
		if = {
			limit = { NOT = { vptl_presidential_transition_qualifies = yes } has_variable = vptl_presidential_history_current_slot }
			set_variable = { name = vptl_presidential_history_departure_reason value = 4 }
			vptl_request_close_presidential_history_episode = yes
		}
		else_if = {
			limit = {
				vptl_presidential_transition_qualifies = yes
				OR = {
					has_variable = vptl_presidential_history_committed_change
					has_variable = vptl_presidential_history_pending_constitutional_accession
					AND = {
						NOT = { in_election_campaign = yes }
						NOT = { has_variable = vptl_presidential_election_transition }
						NOT = { has_variable = vptl_presidential_election_settlement_in_progress }
						NOT = { has_variable = vptl_presidential_inauguration_pending }
						NOT = { has_variable = vptl_presidential_inauguration_in_progress }
						NOT = { has_variable = vptl_presidential_handoff_in_progress }
					}
				}
			}
			if = {
				limit = {
					has_variable = vptl_presidential_history_current_slot
					OR = {
						NOT = { ruler ?= { is_character_alive = yes } }
						NOT = { has_variable = vptl_presidential_history_current_president }
						ruler ?= { NOT = { owner.var:vptl_presidential_history_current_president ?= this } }
					}
				}
				if = { limit = { NOT = { has_variable = vptl_presidential_history_departure_reason } } set_variable = { name = vptl_presidential_history_departure_reason value = 5 } }
				vptl_request_close_presidential_history_episode = yes
			}
			if = {
				limit = { NOT = { has_variable = vptl_presidential_history_current_slot } ruler ?= { is_character_alive = yes } }
				vptl_open_presidential_history_episode = yes
				if = { limit = { has_variable = vptl_presidential_history_current_slot } remove_variable = vptl_presidential_history_pending_constitutional_accession }
			}
		}
		remove_variable = vptl_presidential_history_sync_in_progress
	}
}
