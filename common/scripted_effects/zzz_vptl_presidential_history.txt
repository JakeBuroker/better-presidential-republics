vptl_capture_presidential_history_power_rank = {
	set_variable = { name = vptl_history_capture_rank value = -1 }
	if = { limit = { country_rank = rank_value:decentralized_power } set_variable = { name = vptl_history_capture_rank value = 0 } }
	if = { limit = { country_rank = rank_value:unrecognized_power } set_variable = { name = vptl_history_capture_rank value = 1 } }
	if = { limit = { country_rank = rank_value:insignificant_power } set_variable = { name = vptl_history_capture_rank value = 2 } }
	if = { limit = { country_rank = rank_value:unrecognized_regional_power } set_variable = { name = vptl_history_capture_rank value = 3 } }
	if = { limit = { country_rank = rank_value:minor_power } set_variable = { name = vptl_history_capture_rank value = 4 } }
	if = { limit = { country_rank = rank_value:unrecognized_major_power } set_variable = { name = vptl_history_capture_rank value = 5 } }
	if = { limit = { country_rank = rank_value:major_power } set_variable = { name = vptl_history_capture_rank value = 6 } }
	if = { limit = { country_rank = rank_value:great_power } set_variable = { name = vptl_history_capture_rank value = 7 } }
}

vptl_capture_presidential_history_numeric_rank = {
	# Lifecycle-only prestige ordering. Equal-prestige countries share a rank.
	save_scope_as = vptl_history_rank_subject
	set_variable = { name = vptl_history_capture_score_rank value = 1 }
	every_country = {
		limit = { is_country_alive = yes prestige > scope:vptl_history_rank_subject.prestige }
		scope:vptl_history_rank_subject = { change_variable = { name = vptl_history_capture_score_rank add = 1 } }
	}
}

vptl_prepare_presidential_history_episode_number = {
	# The normal accession path owns the first assignment. The ledger accepts it.
	# Only a former president returning with an older number is advanced here.
	vptl_assign_presidential_order_number_to_current_ruler = yes
	ruler ?= { save_scope_as = vptl_history_numbered_ruler }
	if = {
		limit = { NOT = { has_variable = vptl_presidential_order_counter } scope:vptl_history_numbered_ruler ?= { has_variable = vptl_presidential_order_number } }
		set_variable = { name = vptl_presidential_order_counter value = scope:vptl_history_numbered_ruler.var:vptl_presidential_order_number }
	}
	if = {
		limit = { has_variable = vptl_presidential_history_previous_president has_variable = vptl_presidential_order_counter scope:vptl_history_numbered_ruler ?= { is_character_alive = yes has_variable = vptl_presidential_order_number NOT = { owner.var:vptl_presidential_history_previous_president ?= this } } }
		set_variable = { name = vptl_presidential_history_number_gap value = var:vptl_presidential_order_counter }
		change_variable = { name = vptl_presidential_history_number_gap subtract = scope:vptl_history_numbered_ruler.var:vptl_presidential_order_number }
		if = {
			limit = { var:vptl_presidential_history_number_gap > 0 }
			change_variable = { name = vptl_presidential_order_counter add = 1 }
			scope:vptl_history_numbered_ruler = { set_variable = { name = vptl_presidential_order_number value = owner.var:vptl_presidential_order_counter } vptl_refresh_presidential_office_markers = yes }
		}
		else_if = { limit = { var:vptl_presidential_history_number_gap < 0 } set_variable = { name = vptl_presidential_order_counter value = scope:vptl_history_numbered_ruler.var:vptl_presidential_order_number } }
		remove_variable = vptl_presidential_history_number_gap
	}
}

vptl_capture_presidential_history_episode_data = {
	remove_variable = vptl_history_capture_president
	remove_variable = vptl_history_capture_vice_president
	remove_variable = vptl_history_capture_party
	remove_variable = vptl_history_capture_ig
	set_variable = { name = vptl_history_capture_number value = 0 }
	set_variable = { name = vptl_history_capture_accession_type value = 5 }
	set_variable = { name = vptl_history_capture_gdp_start value = gdp }
	set_variable = { name = vptl_history_capture_population_start value = total_population }
	set_variable = { name = vptl_history_capture_sol_start value = average_sol }
	set_variable = { name = vptl_history_capture_prestige_start value = prestige }
	vptl_capture_presidential_history_power_rank = yes
	vptl_capture_presidential_history_numeric_rank = yes
	vptl_capture_presidential_history_date = yes
	ruler ?= { save_scope_as = vptl_history_capture_president }
	if = {
		limit = { exists = scope:vptl_history_capture_president }
		set_variable = { name = vptl_history_capture_president value = scope:vptl_history_capture_president }
		scope:vptl_history_capture_president = { if = { limit = { is_character_alive = yes NOT = { has_role = character_role_vptl_presidential_history_subject } } add_character_role = character_role_vptl_presidential_history_subject } }
		if = { limit = { scope:vptl_history_capture_president = { has_variable = vptl_presidential_order_number } } set_variable = { name = vptl_history_capture_number value = scope:vptl_history_capture_president.var:vptl_presidential_order_number } }
		if = { limit = { scope:vptl_history_capture_president = { has_variable = vptl_presidential_accession_elected } } set_variable = { name = vptl_history_capture_accession_type value = 1 } }
		else_if = { limit = { scope:vptl_history_capture_president = { has_variable = vptl_presidential_accession_succeeded } } set_variable = { name = vptl_history_capture_accession_type value = 2 } }
		else_if = { limit = { scope:vptl_history_capture_president = { has_variable = vptl_presidential_accession_interim } } set_variable = { name = vptl_history_capture_accession_type value = 3 } }
		else_if = { limit = { scope:vptl_history_capture_president = { has_variable = vptl_presidential_accession_provisional } } set_variable = { name = vptl_history_capture_accession_type value = 4 } }
		scope:vptl_history_capture_president = { this.interest_group ?= { save_scope_as = vptl_history_capture_ig party ?= { save_scope_as = vptl_history_capture_party } } }
	}
	# The first tracked USA episode is Jackson's real presidency, begun March 1829.
	if = { limit = { this = c:USA NOT = { has_variable = vptl_presidential_history_slot_1_populated } scope:vptl_history_capture_president ?= { OR = { has_variable = is_andrew_jackson has_template = usa_andrew_jackson_template } } } set_variable = { name = vptl_history_capture_year value = 1829 } set_variable = { name = vptl_history_capture_month value = 3 } }
	if = { limit = { exists = scope:vptl_history_capture_ig } set_variable = { name = vptl_history_capture_ig value = scope:vptl_history_capture_ig } }
	if = { limit = { exists = scope:vptl_history_capture_party } set_variable = { name = vptl_history_capture_party value = scope:vptl_history_capture_party } }
	if = { limit = { has_variable = vptl_presidential_successor var:vptl_presidential_successor ?= { is_character_alive = yes } } var:vptl_presidential_successor ?= { if = { limit = { NOT = { has_role = character_role_vptl_presidential_history_subject } } add_character_role = character_role_vptl_presidential_history_subject } } set_variable = { name = vptl_history_capture_vice_president value = var:vptl_presidential_successor } }
}

vptl_sync_presidential_history_after_ruler_change = {
	if = { limit = { NOT = { has_variable = vptl_presidential_history_sync_in_progress } } set_variable = vptl_presidential_history_sync_in_progress
		if = { limit = { NOT = { vptl_presidential_transition_qualifies = yes } has_variable = vptl_presidential_history_current_slot } set_variable = { name = vptl_presidential_history_departure_reason value = 6 } vptl_request_close_presidential_history_episode = yes }
		else_if = { limit = { vptl_presidential_transition_qualifies = yes OR = { has_variable = vptl_presidential_history_committed_change has_variable = vptl_presidential_history_pending_constitutional_accession AND = { NOT = { in_election_campaign = yes } NOT = { has_variable = vptl_presidential_election_transition } NOT = { has_variable = vptl_presidential_election_settlement_in_progress } NOT = { has_variable = vptl_presidential_inauguration_pending } NOT = { has_variable = vptl_presidential_inauguration_in_progress } NOT = { has_variable = vptl_presidential_handoff_in_progress } } } }
			if = { limit = { has_variable = vptl_presidential_history_current_slot OR = { NOT = { ruler ?= { is_character_alive = yes } } NOT = { has_variable = vptl_presidential_history_current_president } ruler ?= { NOT = { owner.var:vptl_presidential_history_current_president ?= this } } } } if = { limit = { NOT = { has_variable = vptl_presidential_history_departure_reason } } set_variable = { name = vptl_presidential_history_departure_reason value = 5 } } vptl_request_close_presidential_history_episode = yes }
			if = { limit = { NOT = { has_variable = vptl_presidential_history_current_slot } ruler ?= { is_character_alive = yes } } vptl_open_presidential_history_episode = yes if = { limit = { has_variable = vptl_presidential_history_current_slot } remove_variable = vptl_presidential_history_pending_constitutional_accession } }
		}
		remove_variable = vptl_presidential_history_sync_in_progress
	}
}
