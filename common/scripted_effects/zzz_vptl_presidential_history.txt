vptl_prepare_presidential_history_episode_number = {
	# The normal accession path owns the first assignment. The ledger accepts it.
	# Only a former president returning with an older number is advanced here.
	vptl_assign_presidential_order_number_to_current_ruler = yes
	ruler ?= { save_scope_as = vptl_history_numbered_ruler }
	if = {
		limit = { NOT = { has_variable = vptl_presidential_order_counter } scope:vptl_history_numbered_ruler ?= { has_variable = vptl_presidential_order_number } }
		set_variable = { name = vptl_presidential_order_counter value = scope:vptl_history_numbered_ruler.var:vptl_presidential_order_number }
	}
	if = {
		limit = { has_variable = vptl_presidential_history_previous_president has_variable = vptl_presidential_order_counter scope:vptl_history_numbered_ruler ?= { is_character_alive = yes has_variable = vptl_presidential_order_number NOT = { owner.var:vptl_presidential_history_previous_president ?= this } } }
		set_variable = { name = vptl_presidential_history_number_gap value = var:vptl_presidential_order_counter }
		change_variable = { name = vptl_presidential_history_number_gap subtract = scope:vptl_history_numbered_ruler.var:vptl_presidential_order_number }
		if = {
			limit = { var:vptl_presidential_history_number_gap > 0 }
			change_variable = { name = vptl_presidential_order_counter add = 1 }
			scope:vptl_history_numbered_ruler = { set_variable = { name = vptl_presidential_order_number value = owner.var:vptl_presidential_order_counter } vptl_refresh_presidential_office_markers = yes }
		}
		else_if = { limit = { var:vptl_presidential_history_number_gap < 0 } set_variable = { name = vptl_presidential_order_counter value = scope:vptl_history_numbered_ruler.var:vptl_presidential_order_number } }
		remove_variable = vptl_presidential_history_number_gap
	}
}

vptl_freeze_presidential_history_profile_for_record = {
	# Backfill old records once, and only while their original president resolves.
	if = { limit = { has_variable = $PREFIX$_populated var:$PREFIX$_president ?= { is_character_alive = yes } OR = { NOT = { has_variable = $PREFIX$_culture } NOT = { has_variable = $PREFIX$_religion } NOT = { has_variable = $PREFIX$_ideology } NOT = { has_variable = $PREFIX$_popularity } } }
		var:$PREFIX$_president ?= { save_scope_as = vptl_history_profile_subject culture ?= { save_scope_as = vptl_history_capture_culture } religion ?= { save_scope_as = vptl_history_capture_religion } ideology ?= { save_scope_as = vptl_history_capture_ideology } }
		if = { limit = { NOT = { has_variable = $PREFIX$_culture } exists = scope:vptl_history_capture_culture } set_variable = { name = $PREFIX$_culture value = scope:vptl_history_capture_culture } }
		if = { limit = { NOT = { has_variable = $PREFIX$_religion } exists = scope:vptl_history_capture_religion } set_variable = { name = $PREFIX$_religion value = scope:vptl_history_capture_religion } }
		if = { limit = { NOT = { has_variable = $PREFIX$_ideology } exists = scope:vptl_history_capture_ideology } set_variable = { name = $PREFIX$_ideology value = scope:vptl_history_capture_ideology } }
		if = { limit = { NOT = { has_variable = $PREFIX$_ideology_identity } }
			set_variable = { name = $PREFIX$_ideology_identity value = 0 }
			if = { limit = { var:$PREFIX$_president ?= { has_ideology = ideology:ideology_jacksonian_democrat } } set_variable = { name = $PREFIX$_ideology_identity value = 1 } }
			if = { limit = { var:$PREFIX$_president ?= { has_ideology = ideology:ideology_slaver } } set_variable = { name = $PREFIX$_ideology_identity value = 2 } }
			if = { limit = { var:$PREFIX$_president ?= { has_ideology = ideology:ideology_pacifist } } set_variable = { name = $PREFIX$_ideology_identity value = 3 } }
			if = { limit = { var:$PREFIX$_president ?= { has_ideology = ideology:ideology_republican_leader } } set_variable = { name = $PREFIX$_ideology_identity value = 4 } }
			if = { limit = { var:$PREFIX$_president ?= { has_ideology = ideology:ideology_moderate } } set_variable = { name = $PREFIX$_ideology_identity value = 5 } }
		}
		if = { limit = { NOT = { has_variable = $PREFIX$_popularity } exists = scope:vptl_history_profile_subject } set_variable = { name = $PREFIX$_popularity value = scope:vptl_history_profile_subject.popularity } }
	}
	remove_variable = vptl_history_capture_culture
	remove_variable = vptl_history_capture_religion
	remove_variable = vptl_history_capture_ideology
}

vptl_sync_presidential_history_after_ruler_change = {
	if = { limit = { NOT = { has_variable = vptl_presidential_history_sync_in_progress } } set_variable = vptl_presidential_history_sync_in_progress
		if = { limit = { NOT = { vptl_presidential_transition_qualifies = yes } has_variable = vptl_presidential_history_current_slot } set_variable = { name = vptl_presidential_history_departure_reason value = 6 } vptl_request_close_presidential_history_episode = yes }
		else_if = { limit = { vptl_presidential_transition_qualifies = yes OR = { has_variable = vptl_presidential_history_committed_change has_variable = vptl_presidential_history_pending_constitutional_accession AND = { NOT = { in_election_campaign = yes } NOT = { has_variable = vptl_presidential_election_transition } NOT = { has_variable = vptl_presidential_election_settlement_in_progress } NOT = { has_variable = vptl_presidential_inauguration_pending } NOT = { has_variable = vptl_presidential_inauguration_in_progress } NOT = { has_variable = vptl_presidential_handoff_in_progress } } } }
			if = { limit = { has_variable = vptl_presidential_history_current_slot OR = { NOT = { ruler ?= { is_character_alive = yes } } NOT = { has_variable = vptl_presidential_history_current_president } ruler ?= { NOT = { owner.var:vptl_presidential_history_current_president ?= this } } } } if = { limit = { NOT = { has_variable = vptl_presidential_history_departure_reason } } set_variable = { name = vptl_presidential_history_departure_reason value = 5 } } vptl_request_close_presidential_history_episode = yes }
			if = { limit = { NOT = { has_variable = vptl_presidential_history_current_slot } ruler ?= { is_character_alive = yes } } vptl_open_presidential_history_episode = yes if = { limit = { has_variable = vptl_presidential_history_current_slot } remove_variable = vptl_presidential_history_pending_constitutional_accession } }
		}
		remove_variable = vptl_presidential_history_sync_in_progress
	}
}
