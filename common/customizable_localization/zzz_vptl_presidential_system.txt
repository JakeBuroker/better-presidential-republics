vptl_presidential_constitutional_summary = {
	type = country
	text = {
		trigger = {
			has_law_or_variant = law_type:law_presidential_republic
			has_law_or_variant = law_type:law_autocracy
		}
		localization_key = vptl_presidential_constitutional_summary_concentrated
	}
	text = {
		trigger = {
			has_law_or_variant = law_type:law_presidential_republic
		}
		localization_key = vptl_presidential_constitutional_summary_fixed
	}
	text = {
		trigger = { always = yes }
		localization_key = vptl_presidential_constitutional_summary_default
	}
}

vptl_presidential_successor_office_type = {
	type = country

	text = {
		trigger = {
			vptl_presidential_successor_uses_vice_president_title = yes
		}
		localization_key = vptl_presidential_successor_office_type_true_vp
	}
	text = {
		trigger = {
			OR = {
				c:URU ?= this
			}
		}
		localization_key = vptl_presidential_successor_office_type_legislative
	}
	text = {
		trigger = {
			OR = {
				c:PRG ?= this
				c:HAI ?= this
				c:CHL ?= this
			}
		}
		localization_key = vptl_presidential_successor_office_type_cabinet
	}
	text = {
		trigger = { always = yes }
		localization_key = vptl_presidential_successor_office_type_standin
	}
}

vptl_election_results_presidential_winner_line = {
	type = country

	text = {
		trigger = {
			has_law_or_variant = law_type:law_presidential_republic
			government_transfer_of_power = presidential_elective
			ruler ?= {
				is_character_alive = yes
				has_variable = vptl_presidential_order_number
			}
		}
		localization_key = vptl_election_results_presidential_winner_line_text
	}
	text = {
		trigger = { always = yes }
		localization_key = vptl_election_results_presidential_winner_line_blank
	}
}

vptl_presidential_accession_type = {
	type = character
	text = {
		trigger = {
			has_variable = vptl_presidential_accession_interim
		}
		localization_key = vptl_presidential_accession_type_interim
	}
	text = {
		trigger = {
			has_variable = vptl_presidential_accession_provisional
		}
		localization_key = vptl_presidential_accession_type_provisional
	}
	text = {
		trigger = {
			has_variable = vptl_presidential_accession_initial
		}
		localization_key = vptl_presidential_accession_type_initial
	}
	text = {
		trigger = {
			OR = {
				has_variable = vptl_presidential_accession_succeeded
				has_variable = vptl_presidential_succeeded_to_office
			}
		}
		localization_key = vptl_presidential_accession_type_succeeded
	}
	text = {
		trigger = {
			OR = {
				has_variable = vptl_presidential_accession_elected
				has_variable = vptl_presidential_terms_served
			}
		}
		localization_key = vptl_presidential_accession_type_elected
	}
	text = {
		trigger = { always = yes }
		localization_key = vptl_presidential_accession_type_unrecorded
	}
}
