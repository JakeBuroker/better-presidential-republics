on_game_started_after_lobby = {
	on_actions = {
		vptl_presidential_term_limits_startup
	}
}

# On-actions are Victoria 3 event hooks. They call BPR effects at lifecycle events.
# Keep the hooks small; detailed work belongs in scripted effects and triggers.

on_monthly_pulse_country = {
	on_actions = {
		vptl_presidential_term_limits_monthly
		vptl_presidential_inauguration_cleanup_monthly
	}
}

on_monthly_pulse_country_elections = {
	on_actions = {
		vptl_presidential_term_limits_monthly_elections
	}
}

on_election_campaign_start = {
	on_actions = {
		vptl_presidential_term_limits_election_start
	}
}

on_election_campaign_end = {
	on_actions = {
		vptl_presidential_term_limits_election_end
	}
}

on_ruler_selected = {
	on_actions = {
		vptl_presidential_successor_ruler_selected
	}
}

on_character_death = {
	on_actions = {
		vptl_presidential_character_death
	}
}

# A law enters presidential history only after the game confirms activation.
# ROOT is the activated law; owner is the country whose law changed.
on_law_activated = {
	on_actions = {
		vptl_presidential_history_law_activated
	}
}

vptl_presidential_history_law_activated = {
	effect = {
		owner ?= {
			if = {
				limit = {
					vptl_presidential_transition_qualifies = yes
					has_variable = vptl_presidential_history_current_slot
					has_variable = vptl_presidential_history_current_president
					ruler ?= { owner.var:vptl_presidential_history_current_president ?= this }
				}
				set_variable = { name = vptl_presidential_history_pending_law value = ROOT }
				vptl_record_law_for_active_presidential_history = yes
			}
		}
	}
}

vptl_presidential_term_limits_startup = {
	effect = {
		every_country = {
			limit = {
				OR = {
					vptl_presidential_transition_qualifies = yes
					has_variable = vptl_presidential_inauguration_pending
				}
			}
			if = {
				limit = { vptl_presidential_transition_qualifies = yes }
				vptl_initialize_presidential_term_limits = yes
				vptl_repair_presidential_campaign_ticket_if_needed = yes
			}
			else = { vptl_cleanup_presidential_inauguration_transition = yes }
		}
	}
}

vptl_presidential_term_limits_monthly = {
	trigger = {
		has_law_or_variant = law_type:law_presidential_republic
		government_transfer_of_power = presidential_elective
		NOT = { has_law_or_variant = law_type:law_single_party_state }
	}
	effect = {
		vptl_maintain_presidential_inauguration_transition = yes
		if = {
			limit = {
				in_election_campaign = yes
				has_variable = vptl_presidential_debug_logging
			}
			debug_log = "BPR checkpoint=campaign_tick [TimeKeeper.GetCurrentDate.GetString] country=[THIS.GetCountry.GetNameNoFormatting]"
			vptl_debug_log_presidential_state = yes
		}
		vptl_repair_presidential_campaign_ticket_if_needed = yes
		vptl_finalize_presidential_election_transition_if_campaign_ended = yes
		vptl_promote_presidential_successor_if_ruler_is_dead = yes
		vptl_promote_presidential_successor_if_tracked_ruler_is_dead = yes
		if = {
			limit = {
				NOT = { has_variable = vptl_presidential_election_transition }
			}
			vptl_replace_ineligible_usa_presidential_ruler_with_successor = yes
		}
		vptl_correct_presidential_ruler_change_during_campaign = yes
		vptl_correct_unexpected_presidential_ruler_change = yes
		vptl_initialize_presidential_term_limits = yes
		vptl_record_presidential_service_time = yes
		vptl_sync_presidential_history_after_ruler_change = yes
	}
}

vptl_presidential_inauguration_cleanup_monthly = {
	trigger = {
		OR = {
			has_variable = vptl_presidential_inauguration_pending
			has_variable = vptl_presidential_history_current_slot
		}
		NOT = { vptl_presidential_transition_qualifies = yes }
	}
	effect = {
		vptl_maintain_presidential_inauguration_transition = yes
		vptl_sync_presidential_history_after_ruler_change = yes
	}
}

vptl_presidential_successor_ruler_selected = {
	effect = {
		root.owner = {
			if = {
				limit = { has_variable = vptl_presidential_debug_logging }
				debug_log = "BPR checkpoint=on_ruler_selected [TimeKeeper.GetCurrentDate.GetString] country=[THIS.GetCountry.GetNameNoFormatting]"
				vptl_debug_log_presidential_state = yes
			}
			if = {
				limit = {
					OR = {
						has_variable = vptl_presidential_election_settlement_in_progress
						has_variable = vptl_presidential_inauguration_in_progress
					}
				}
				vptl_sync_presidential_current_ruler = yes
			}
			else_if = {
				limit = {
					has_variable = vptl_presidential_election_transition
					in_election_campaign = no
				}
				vptl_settle_presidential_election = yes
			}
			else_if = {
				limit = { has_variable = vptl_presidential_inauguration_pending }
				if = {
					limit = {
						has_variable = vptl_presidential_handoff_in_progress
						has_variable = vptl_presidential_handoff_target
						has_variable = vptl_presidential_pending_death_handling
						ruler ?= { owner.var:vptl_presidential_handoff_target ?= this }
					}
					vptl_sync_presidential_current_ruler = yes
					set_variable = { name = vptl_presidential_transition_outgoing_ruler value = ruler }
				}
				else_if = {
					limit = {
						has_variable = vptl_presidential_transition_outgoing_ruler
						var:vptl_presidential_transition_outgoing_ruler ?= { is_character_alive = yes }
						ruler ?= {
							NOT = { owner.var:vptl_presidential_transition_outgoing_ruler ?= this }
						}
					}
					vptl_restore_transition_outgoing_ruler = yes
				}
				else_if = {
					limit = {
						has_variable = vptl_presidential_current_ruler
						var:vptl_presidential_current_ruler ?= { is_character_alive = yes }
						ruler ?= { NOT = { owner.var:vptl_presidential_current_ruler ?= this } }
					}
					vptl_restore_transition_acting_ruler = yes
				}
				else = {
					vptl_sync_presidential_current_ruler = yes
					set_variable = { name = vptl_presidential_transition_outgoing_ruler value = ruler }
				}
				vptl_refresh_lame_duck_administration = yes
			}
			else = {
				vptl_correct_presidential_ruler_change_during_campaign = yes
				vptl_promote_presidential_successor_if_ruler_is_dead = yes
				vptl_promote_presidential_successor_if_tracked_ruler_is_dead = yes
				vptl_apply_presidential_successor_if_current_ruler_died_during_campaign = yes
				vptl_enforce_presidential_successor_for_non_election_transition = yes
			}
			vptl_sync_presidential_history_after_ruler_change = yes
		}
	}
}

vptl_presidential_character_death = {
	effect = {
		save_scope_as = vptl_presidential_dead_character
		owner = {
			save_scope_as = vptl_presidential_death_owner_country
			if = {
				limit = {
					has_variable = vptl_presidential_history_current_president
					has_variable = vptl_presidential_history_current_slot
					var:vptl_presidential_history_current_president ?= { this = scope:vptl_presidential_dead_character }
				}
				vptl_latch_presidential_history_death_rank = yes
			}
			if = {
				limit = { has_variable = vptl_presidential_inauguration_pending }
				vptl_handle_presidential_inauguration_character_death = yes
			}
			else = {
				vptl_refresh_presidential_successor_if_ruler_is_stable = yes
				vptl_promote_presidential_successor_if_ruler_is_dead = yes
				vptl_promote_presidential_successor_if_tracked_ruler_is_dead = yes
				vptl_apply_presidential_successor_if_current_ruler_died_during_campaign = yes
				vptl_refresh_presidential_successor_if_ruler_is_stable = yes
			}
			vptl_promote_running_mate_for_dead_presidential_ticket_head = yes
			vptl_rebuild_presidential_campaign_ticket = yes
			vptl_refresh_presidential_successor_display_marker = yes
			if = {
				limit = {
					has_variable = vptl_presidential_history_current_president
					var:vptl_presidential_history_current_president ?= { this = scope:vptl_presidential_dead_character }
				}
				set_variable = { name = vptl_presidential_history_departure_reason value = 1 }
				set_variable = vptl_presidential_history_committed_change
				set_variable = vptl_presidential_history_pending_constitutional_accession
			}
			vptl_sync_presidential_history_after_ruler_change = yes
			remove_variable = vptl_presidential_history_committed_change
			if = {
				limit = { has_variable = vptl_presidential_history_current_slot }
				remove_variable = vptl_presidential_history_pending_constitutional_accession
			}
		}
		if = {
			limit = {
				exists = home_country
				home_country = {
					NOT = { this = scope:vptl_presidential_death_owner_country }
				}
			}
			home_country = {
				vptl_refresh_presidential_successor_if_ruler_is_stable = yes
				vptl_promote_running_mate_for_dead_presidential_ticket_head = yes
				vptl_rebuild_presidential_campaign_ticket = yes
				vptl_refresh_presidential_successor_display_marker = yes
			}
		}
	}
}
