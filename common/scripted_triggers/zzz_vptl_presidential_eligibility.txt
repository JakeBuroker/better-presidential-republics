vptl_presidential_candidate_meets_age_requirement = {
	OR = {
		NOT = {
			owner = { c:USA ?= this }
		}
		age >= 35
	}
}

vptl_usa_presidential_origin_explicitly_blocked = {
	OR = {
		has_variable = vptl_usa_presidential_origin_ineligible
		has_variable = vptl_presidential_origin_ineligible
	}
}

vptl_usa_presidential_origin_explicitly_allowed = {
	OR = {
		has_variable = vptl_usa_natural_born_presidential_eligible
		has_variable = vptl_presidential_origin_eligible
		has_variable = vptl_usa_founder_presidential_eligible
	}
}

vptl_usa_presidential_origin_generated_home_country_eligible = {
	AND = {
		NOT = { is_historical = yes }
		exists = home_country
		home_country = {
			exists = c:USA
			c:USA = this
		}
	}
}

vptl_usa_presidential_origin_historical_fallback_eligible = {
	AND = {
		is_historical = yes
		OR = {
			culture = cu:yankee
			culture = cu:dixie
			culture = cu:afro_american
		}
	}
}

vptl_usa_presidential_origin_eligible = {
	OR = {
		NOT = {
			owner = { c:USA ?= this }
		}
		AND = {
			owner = { c:USA ?= this }
			NOT = { vptl_usa_presidential_origin_explicitly_blocked = yes }
			OR = {
				vptl_usa_presidential_origin_explicitly_allowed = yes
				vptl_usa_presidential_origin_generated_home_country_eligible = yes
				vptl_usa_presidential_origin_historical_fallback_eligible = yes
			}
		}
	}
}

# These triggers are reusable yes/no conditions for candidates and successors.
# Character triggers run on a character scope; owner points back to that character's country.

vptl_presidential_basic_eligibility = {
	vptl_presidential_candidate_meets_age_requirement = yes
	vptl_usa_presidential_origin_eligible = yes
}

vptl_presidential_candidate_eligible = {
	vptl_presidential_basic_eligibility = yes
	is_character_alive = yes
	has_role_of_type = politician
	is_heir_of_own_country = no
	NOT = {
		AND = {
			has_variable = vptl_presidential_terms_served
			var:vptl_presidential_terms_served >= 2
		}
	}
	NOT = { has_variable = vptl_presidential_term_limited }
}

vptl_presidential_new_successor_candidate_eligible = {
	vptl_presidential_basic_eligibility = yes
	is_character_alive = yes
	has_role_of_type = politician
	is_ruler_of_own_country = no
	is_heir_of_own_country = no
	NOT = { has_role = character_role_vptl_former_president }
	NOT = { owner.var:vptl_presidential_ticket_candidate ?= this }
	NOT = { owner.var:vptl_presidential_opposition_candidate ?= this }
	NOT = { owner.var:vptl_presidential_final_elected_ruler ?= this }
	NOT = {
		AND = {
			has_variable = vptl_vice_presidential_terms_served
			var:vptl_vice_presidential_terms_served >= 2
		}
	}
}

# A saved running mate may remain on their existing ticket after succeeding to the presidency mid-campaign.
vptl_presidential_existing_campaign_running_mate_eligible = {
	OR = {
		vptl_presidential_new_successor_candidate_eligible = yes
		AND = {
			vptl_presidential_basic_eligibility = yes
			is_character_alive = yes
			has_role_of_type = politician
			is_ruler_of_own_country = yes
			is_heir_of_own_country = no
			NOT = { has_role = character_role_vptl_former_president }
			NOT = { owner.var:vptl_presidential_ticket_candidate ?= this }
			NOT = { owner.var:vptl_presidential_opposition_candidate ?= this }
			NOT = { owner.var:vptl_presidential_final_elected_ruler ?= this }
			NOT = {
				AND = {
					has_variable = vptl_vice_presidential_terms_served
					var:vptl_vice_presidential_terms_served >= 2
				}
			}
			owner = {
				has_variable = vptl_presidential_election_transition
				has_variable = vptl_presidential_death_succession_lock
				has_variable = vptl_presidential_death_successor
			}
			owner.var:vptl_presidential_death_successor ?= this
			OR = {
				owner.var:vptl_presidential_ticket_running_mate ?= this
				owner.var:vptl_presidential_opposition_running_mate ?= this
			}
		}
	}
}

# Party-owned tickets may preserve the exact saved running mate after that character
# constitutionally succeeds during the campaign. New searches still use the ordinary
# non-ruler successor trigger.
vptl_presidential_party_existing_running_mate_eligible = {
	OR = {
		vptl_presidential_new_successor_candidate_eligible = yes
		AND = {
			vptl_presidential_basic_eligibility = yes
			is_character_alive = yes
			has_role_of_type = politician
			is_ruler_of_own_country = yes
			is_heir_of_own_country = no
			NOT = { has_role = character_role_vptl_former_president }
			NOT = {
				AND = {
					has_variable = vptl_vice_presidential_terms_served
					var:vptl_vice_presidential_terms_served >= 2
				}
			}
			owner = {
				has_variable = vptl_presidential_election_transition
				has_variable = vptl_presidential_death_succession_lock
				has_variable = vptl_presidential_death_successor
			}
			owner.var:vptl_presidential_death_successor ?= this
		}
	}
}

# Election settlement may temporarily make the authoritative winning running mate the vanilla ruler.
vptl_presidential_certified_running_mate_eligible = {
	vptl_presidential_basic_eligibility = yes
	is_character_alive = yes
	has_role_of_type = politician
	OR = {
		is_ruler_of_own_country = no
		AND = {
			is_ruler_of_own_country = yes
			owner = {
				has_variable = vptl_presidential_election_settlement_in_progress
				has_variable = vptl_presidential_vanilla_selected_ruler
			}
			owner.var:vptl_presidential_vanilla_selected_ruler ?= this
		}
		AND = {
			is_ruler_of_own_country = yes
			owner = {
				has_variable = vptl_presidential_election_settlement_in_progress
				has_variable = vptl_presidential_election_transition
				has_variable = vptl_presidential_death_succession_lock
				has_variable = vptl_presidential_death_successor
			}
			owner.var:vptl_presidential_death_successor ?= this
			OR = {
				owner.var:vptl_presidential_ticket_running_mate ?= this
				owner.var:vptl_presidential_opposition_running_mate ?= this
				owner.var:vptl_presidential_winning_ticket_party ?= {
					has_variable = vptl_presidential_party_ticket_running_mate
					var:vptl_presidential_party_ticket_running_mate ?= {
						owner.var:vptl_presidential_death_successor ?= this
					}
				}
			}
		}
	}
	is_heir_of_own_country = no
	NOT = { has_role = character_role_vptl_former_president }
	NOT = { owner.var:vptl_presidential_ticket_candidate ?= this }
	NOT = { owner.var:vptl_presidential_opposition_candidate ?= this }
	NOT = { owner.var:vptl_presidential_final_elected_ruler ?= this }
	NOT = { owner.var:vptl_president_elect ?= this }
	NOT = {
		AND = {
			has_variable = vptl_vice_presidential_terms_served
			var:vptl_vice_presidential_terms_served >= 2
		}
	}
}

# The authoritative certified running mate remains valid if they are also the outgoing or acting president.
vptl_presidential_certified_outgoing_vice_president_elect_eligible = {
	vptl_presidential_basic_eligibility = yes
	is_character_alive = yes
	has_role_of_type = politician
	is_heir_of_own_country = no
	owner = {
		has_variable = vptl_presidential_inauguration_pending
		has_variable = vptl_president_elect
		has_variable = vptl_vice_president_elect
		OR = {
			has_variable = vptl_presidential_transition_outgoing_ruler
			has_variable = vptl_presidential_final_running_mate
		}
	}
	owner.var:vptl_vice_president_elect ?= this
	OR = {
		AND = {
			owner = { has_variable = vptl_presidential_transition_outgoing_ruler }
			owner.var:vptl_presidential_transition_outgoing_ruler ?= this
		}
		AND = {
			owner = { has_variable = vptl_presidential_final_running_mate }
			owner.var:vptl_presidential_final_running_mate ?= this
		}
	}
	NOT = { owner.var:vptl_president_elect ?= this }
	NOT = {
		AND = {
			has_variable = vptl_vice_presidential_terms_served
			var:vptl_vice_presidential_terms_served >= 2
		}
	}
}

vptl_presidential_campaign_ticket_valid_legacy = {
	has_variable = vptl_presidential_election_transition
	has_variable = vptl_presidential_ticket_candidate
	var:vptl_presidential_ticket_candidate ?= {
		vptl_presidential_candidate_eligible = yes
		NOT = { owner.var:vptl_presidential_ticket_running_mate ?= this }
		NOT = { owner.var:vptl_presidential_opposition_candidate ?= this }
		NOT = { owner.var:vptl_presidential_opposition_running_mate ?= this }
	}
	OR = {
		NOT = { has_variable = vptl_presidential_ticket_running_mate }
		var:vptl_presidential_ticket_running_mate ?= {
			vptl_presidential_existing_campaign_running_mate_eligible = yes
			NOT = { owner.var:vptl_presidential_ticket_candidate ?= this }
			NOT = { owner.var:vptl_presidential_opposition_candidate ?= this }
			NOT = { owner.var:vptl_presidential_opposition_running_mate ?= this }
		}
	}
	OR = {
		NOT = { has_variable = vptl_presidential_opposition_candidate }
		var:vptl_presidential_opposition_candidate ?= {
			vptl_presidential_candidate_eligible = yes
			is_ruler_of_own_country = no
			NOT = { owner.var:vptl_presidential_ticket_candidate ?= this }
			NOT = { owner.var:vptl_presidential_ticket_running_mate ?= this }
			NOT = { owner.var:vptl_presidential_opposition_running_mate ?= this }
		}
	}
	OR = {
		NOT = { has_variable = vptl_presidential_opposition_running_mate }
		var:vptl_presidential_opposition_running_mate ?= {
			vptl_presidential_existing_campaign_running_mate_eligible = yes
			NOT = { owner.var:vptl_presidential_ticket_candidate ?= this }
			NOT = { owner.var:vptl_presidential_ticket_running_mate ?= this }
			NOT = { owner.var:vptl_presidential_opposition_candidate ?= this }
		}
	}
}
